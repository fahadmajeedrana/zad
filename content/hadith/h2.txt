## اَلْبَابُ الْأَوَّلُ: اَلْاِعْتِقَادَاتُ — اعتقادات
# پہلی فصل: توحید
1|عَنْ مُعَاذِ بْنِ جَبَلٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ لِيْ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَفَاتِيْحُ الْجَنَّةِ شَهَادَةُ أَنْ لَا إِلٰهَ إِلَّا اللّٰهُ.|مسند احمد: رقم الحدیث 22001|مَفَاتِيْحُ=کنجیاں~الْجَنَّةِ=جنت~شَهَادَةُ=گواہی دینا~أَنْ=کہ~لَا=نہیں~إِلٰهَ=معبود~إِلَّا=مگر~اللّٰهُ=اللہ|حضرت معاذ بن جبل رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے مجھ سے ارشاد فرمایا کہ جنت کی کنجیاں لا الہ الا اللہ کی گواہی دینا ہے۔
2|عَنْ عُثْمَانَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ مَاتَ وَهُوَ يَعْلَمُ أَنَّهٗ لَا إِلٰهَ إِلَّا اللّٰهُ دَخَلَ الْجَنَّةَ.|صحیح مسلم: رقم الحدیث 136|مَنْ=جو شخص~مَاتَ=مر~يَعْلَمُ=یقین رکھتا ہے~أَنَّهٗ=بلاشبہ یہ بات~لَا=نہیں~إِلٰهَ=معبود~إِلَّا=مگر~اللّٰهُ=اللہ~دَخَلَ=وہ داخل ہوگا~الْجَنَّةَ=جنت|حضرت عثمان بن عفان رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جو شخص اس حالت میں فوت ہوا کہ وہ یقین رکھتا تھا کہ اللہ تعالیٰ کے سوا کوئی عبادت کے لائق نہیں ہے تو وہ جنت میں جائے گا۔
3|عَنِ ابْنِ عُمَرَ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: بُنِيَ الْإِسْلَامُ عَلٰى خَمْسٍ شَهَادَةِ أَنْ لَا إِلٰهَ إِلَّا اللّٰهُ وَأَنَّ مُحَمَّدًا رَسُوْلُ اللّٰهِ وَإِقَامِ الصَّلَاةِ وَإِيْتَاءِ الزَّكَاةِ وَالْحَجِّ وَصَوْمِ رَمَضَانَ.|صحیح البخاری: رقم الحدیث 8|بُنِيَ=بنیاد رکھی گئی~الْإِسْلَامُ=اسلام~عَلٰى=پر~خَمْسٍ=پانچ~شَهَادَةِ=گواہی دینا~أَنْ=کہ~لَا=نہیں~إِلٰهَ=معبود~إِلَّا=مگر~اللّٰهُ=اللہ~أَنَّ=بے شک~مُحَمَّدًا رَسُوْلُ اللّٰهِ=محمد اللہ تعالیٰ کے رسول ہیں~إِقَامِ=قائم کرنا~الصَّلَاةِ=نماز~إِيْتَاءِ=ادا کرنا~الزَّكَاةِ=زکوٰۃ~الْحَجِّ=حج کرنا~صَوْمِ=روزہ رکھنا~رَمَضَانَ=ماہ رمضان|حضرت عبد اللہ بن عمر رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: اسلام کی بنیاد پانچ ستونوں پر رکھی گئی ہے۔ اس بات کی گواہی دینا کہ اللہ تعالیٰ کے سوا کوئی عبادت کے لائق نہیں ہے اور حضرت محمد صلی اللہ علیہ وسلم اللہ کے رسول ہیں، نماز قائم کرنا، زکوٰۃ ادا کرنا، حج کرنا اور رمضان کے روزے رکھنا۔
# دوسری فصل: رسالت
1|عَنْ عُبَادَةَ بْنِ الصَّامِتِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: مَنْ شَهِدَ أَنْ لَا إِلٰهَ إِلَّا اللّٰهُ وَأَنَّ مُحَمَّدًا رَسُوْلُ اللّٰهِ حَرَّمَ اللّٰهُ عَلَيْهِ النَّارَ.|سنن الترمذی: رقم الحدیث 2638|مَنْ=جو شخص~شَهِدَ=اس نے گواہی دی~حَرَّمَ=حرام کر دی~النَّارَ=آگ|حضرت عبادہ بن صامت رضی اللہ عنہ سے روایت ہے کہ میں نے رسول اللہ صلی اللہ علیہ وسلم سے سنا ہے کہ جو آدمی گواہی دے کہ اللہ تعالیٰ کے سوا کوئی عبادت کے لائق نہیں اور محمد صلی اللہ علیہ وسلم اس کے رسول ہیں تو اللہ تعالیٰ اس شخص پر دوزخ کی آگ حرام کر دیتے ہیں۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَا يُؤْمِنُ أَحَدُكُمْ حَتّٰى أَكُوْنَ أَحَبَّ إِلَيْهِ مِنْ وَالِدِهٖ وَوَلَدِهٖ وَالنَّاسِ أَجْمَعِيْنَ.|صحیح البخاری: رقم الحدیث 15|لَا يُؤْمِنُ=مومن نہیں ہو سکتا~أَحَدُكُمْ=تم میں سے کوئی~حَتّٰى=یہاں تک کہ~أَكُوْنَ=میں ہو جاؤں~أَحَبَّ=سب سے زیادہ محبوب~إِلَيْهِ=اسے~مِنْ=سے~وَالِدِهٖ=اس کے والد~وَلَدِهٖ=اس کی اولاد~النَّاسِ=لوگ~أَجْمَعِيْنَ=تمام|حضرت انس رضی اللہ عنہ سے روایت ہے کہ رسول اکرم صلی اللہ علیہ وسلم نے فرمایا: تم میں سے کوئی شخص اس وقت تک کامل مومن نہیں ہو سکتا جب تک اس کو اپنے ماں باپ، اپنی اولاد اور سب لوگوں سے زیادہ مجھ سے محبت نہ ہو۔
3|عَنْ عَبْدِ اللّٰهِ بْنِ عَمْرٍو رَضِيَ اللّٰهُ عَنْهُمَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَا يُؤْمِنُ أَحَدُكُمْ حَتّٰى يَكُوْنَ هَوَاهُ تَبَعًا لِمَا جِئْتُ بِهٖ.|شرح السنۃ للبغوی: رقم الحدیث 104|لَا يُؤْمِنُ=مومن نہیں ہو سکتا~أَحَدُكُمْ=تم میں سے کوئی~حَتّٰى=یہاں تک کہ~يَكُوْنَ=ہو جائے~هَوَاهُ=اس کی خواہش~تَبَعًا=تابع~لِمَا=اس چیز کے~جِئْتُ بِهٖ=جسے میں لایا|حضرت عبد اللہ بن عمرو رضی اللہ عنہما سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: تم میں سے کوئی شخص مومن نہیں ہو سکتا جب تک کہ اس کی خواہشات میرے لائے ہوئے دین کے تابع نہ ہو جائیں۔
# تیسری فصل: ختم نبوت
1|عَنْ ثَوْبَانَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَنَا خَاتَمُ النَّبِيِّيْنَ لَا نَبِيَّ بَعْدِيْ.|سنن ابی داؤد: رقم الحدیث 4252|أَنَا=میں~خَاتَمُ=ختم کرنے والا~النَّبِيِّيْنَ=نبیوں~لَا=نہیں~نَبِيَّ=نبی~بَعْدِيْ=میرے بعد|حضرت ثوبان رضی اللہ عنہ فرماتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: میں آخری نبی ہوں میرے بعد کوئی نبی پیدا نہیں ہوگا۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: فُضِّلْتُ عَلَى الْأَنْبِيَاءِ بِسِتٍّ: أُعْطِيْتُ جَوَامِعَ الْكَلِمِ وَنُصِرْتُ بِالرُّعْبِ وَأُحِلَّتْ لِيَ الْغَنَائِمُ وَجُعِلَتْ لِيَ الْأَرْضُ طَهُوْرًا وَمَسْجِدًا وَأُرْسِلْتُ إِلَى الْخَلْقِ كَافَّةً وَخُتِمَ بِيَ النَّبِيُّوْنَ.|صحیح مسلم: رقم الحدیث 1167|فُضِّلْتُ=مجھے فضیلت دی گئی~عَلٰى=پر~الْأَنْبِيَاءِ=تمام انبیاء~بِسِتٍّ=چھ کے ساتھ~أُعْطِيْتُ=مجھے دی گئیں~جَوَامِعَ=جامع~الْكَلِمِ=کلمات~نُصِرْتُ=میری مدد کی گئی~بِالرُّعْبِ=رعب کے ساتھ~أُحِلَّتْ=حلال کی گئی~لِيْ=میرے لیے~الْغَنَائِمُ=غنیمتیں~جُعِلَتْ=بنا دی گئی~الْأَرْضُ=زمین~طَهُوْرًا=پاکی کا ذریعہ~مَسْجِدًا=سجدہ گاہ~أُرْسِلْتُ=مجھے بھیجا گیا~إِلَى=طرف~الْخَلْقِ=مخلوق~كَافَّةً=تمام~خُتِمَ=ختم کر دیا گیا~بِيْ=مجھ پر~النَّبِيُّوْنَ=نبیوں|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: چھ چیزیں عطا کر کے مجھے بقیہ انبیاء علیہم السلام پر فضیلت دی گئی ہے: مجھے جامع کلمات عطا کیے گئے ہیں، رعب عطا کر کے میری مدد فرمائی گئی ہے، میرے لیے مال غنیمت کو حلال کیا گیا ہے، پوری زمین کو میرے لیے پاک کرنے والی اور سجدہ گاہ بنا دیا گیا ہے اور مجھے ساری مخلوق کا نبی بنا کر بھیجا گیا ہے اور مجھ پر سلسلہ نبوت ختم کر دیا گیا ہے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِنَّ مَثَلِيْ وَمَثَلَ الْأَنْبِيَاءِ مِنْ قَبْلِيْ كَمَثَلِ رَجُلٍ بَنٰى بَيْتًا فَأَحْسَنَهٗ وَأَجْمَلَهٗ إِلَّا مَوْضِعَ لَبِنَةٍ مِنْ زَاوِيَةٍ فَجَعَلَ النَّاسُ يَطُوْفُوْنَ بِهٖ وَيَعْجَبُوْنَ لَهٗ وَيَقُوْلُوْنَ هَلَّا وُضِعَتْ هٰذِهِ اللَّبِنَةُ قَالَ فَأَنَا اللَّبِنَةُ وَأَنَا خَاتَمُ النَّبِيِّيْنَ.|صحیح البخاری: رقم الحدیث 3535|مَثَلِيْ=میری مثال~مَثَلَ=مثال~الْأَنْبِيَاءِ=نبیوں~مِنْ قَبْلِيْ=مجھ سے پہلے~كَمَثَلِ=کی طرح~رَجُلٍ=آدمی~بَنٰى=اس نے تعمیر کیا~بَيْتًا=گھر~فَأَحْسَنَهٗ=پھر اسے خوب اچھا بنایا~أَجْمَلَهٗ=اور اسے خوبصورت بنایا~إِلَّا=مگر~مَوْضِعَ=جگہ~لَبِنَةٍ=ایک اینٹ~مِنْ=سے~زَاوِيَةٍ=کونہ~فَجَعَلَ=پھر شروع کر دیا~النَّاسُ=لوگ~يَطُوْفُوْنَ=چکر لگاتے ہیں~يَعْجَبُوْنَ لَهٗ=وہ انہیں اچھا لگتا ہے~فَيَقُوْلُوْنَ=چنانچہ وہ کہنے لگے~هَلَّا=کیوں نہیں~وُضِعَتْ=رکھی گئی~اللَّبِنَةُ=ایک اینٹ~أَنَا=میں~خَاتَمُ=ختم کرنے والا~النَّبِيِّيْنَ=نبی کی جمع|حضرت ابوہریرہ رضی اللہ عنہ کہتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: میری اور دوسرے تمام انبیاء کی مثال اس گھر کی سی ہے جسے ایک شخص نے بنایا۔ اس نے اس گھر کو انتہائی حسین و جمیل بنایا اور مکمل کیا لیکن اس کے ایک کونے میں ایک اینٹ کی جگہ چھوڑ دی۔ لوگ عمارت کو دیکھنے کے لیے اس کے ارد گرد چکر لگاتے ہیں اور عمارت انہیں اچھی لگتی ہے تو وہ کہتے ہیں: ایک اینٹ یہاں کیوں نہ رکھی گئی تاکہ عمارت مکمل ہو جاتی۔ آپ صلی اللہ علیہ وسلم نے فرمایا: وہ اینٹ میں ہوں اور میں آخری نبی ہوں۔
# چوتھی فصل: تقدیر
1|عَنْ عَلِيٍّ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَا يُؤْمِنُ عَبْدٌ حَتّٰى يُؤْمِنَ بِأَرْبَعٍ: بِاللّٰهِ وَحْدَهٗ لَا شَرِيْكَ لَهٗ وَأَنِّيْ رَسُوْلُ اللّٰهِ وَبِالْبَعْثِ بَعْدَ الْمَوْتِ وَالْقَدَرِ.|سنن ابن ماجہ: رقم الحدیث 81|لَا يُؤْمِنُ=مومن نہیں ہو سکتا~عَبْدٌ=آدمی~حَتّٰى=یہاں تک کہ~يُؤْمِنَ=ایمان لے آئے~بِأَرْبَعٍ=چار چیزوں پر~بِاللّٰهِ وَحْدَهٗ لَا شَرِيْكَ لَهٗ=اللہ وحدہ لا شریک پر~أَنِّيْ رَسُوْلُ اللّٰهِ=اور مجھے اللہ کا رسول ماننے پر~بِالْبَعْثِ=اور اٹھنے پر~بَعْدَ الْمَوْتِ=موت کے بعد~الْقَدَرِ=تقدیر|حضرت علی رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: کوئی آدمی اس وقت تک مومن نہیں ہو سکتا جب تک چار چیزوں پر ایمان نہ لے آئے: اللہ تعالیٰ کی توحید پر جس کا کوئی شریک نہیں، میری رسالت پر، مرنے کے بعد دوبارہ اٹھائے جانے پر اور تقدیر پر۔
2|حَدَّثَنَا يَحْيَى بْنُ عَبْدِ اللّٰهِ بْنِ أَبِيْ مُلَيْكَةَ عَنْ أَبِيْهِ أَنَّهٗ دَخَلَ عَلٰى عَائِشَةَ فَذَكَرَ لَهَا شَيْئًا مِنَ الْقَدَرِ فَقَالَتْ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: مَنْ تَكَلَّمَ فِيْ شَيْءٍ مِنَ الْقَدَرِ سُئِلَ عَنْهُ يَوْمَ الْقِيَامَةِ وَمَنْ لَمْ يَتَكَلَّمْ فِيْهِ لَمْ يُسْأَلْ عَنْهُ.|سنن ابن ماجہ: رقم الحدیث 84|مَنْ=جس شخص~تَكَلَّمَ=اس نے کلام کیا~فِيْ=میں~شَيْءٍ=کسی چیز~مِنَ الْقَدَرِ=تقدیر میں سے~سُئِلَ=سوال کیا جائے گا~عَنْهُ=اس سے~يَوْمَ الْقِيَامَةِ=قیامت کے دن~لَمْ يَتَكَلَّمْ=کلام نہیں کیا~فِيْهِ=اس میں~لَمْ يُسْأَلْ=سوال نہیں کیا جائے گا|حضرت یحییٰ بن عبد اللہ بن ابی ملیکہ اپنے والد سے روایت ہیں کہ وہ حضرت عائشہ صدیقہ رضی اللہ عنہا کی خدمت میں حاضر ہوئے اور ان سے تقدیر کے بارے میں کوئی بات کی تو انہوں (حضرت عائشہ رضی اللہ عنہا) نے کہا کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو فرماتے ہوئے سنا ہے: جس آدمی نے تقدیر کے مسئلہ میں بحث و مباحثہ کیا تو قیامت کے دن اس سے اس کے بارے میں باز پرس ہوگی اور جس نے اس مسئلہ میں گفتگو نہ کی اس سے کوئی سوال نہیں ہوگا۔
3|عَنْ عُمَرَ بْنِ الْخَطَّابِ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَا تُجَالِسُوْا أَهْلَ الْقَدَرِ وَلَا تُفَاتِحُوْهُمْ.|سنن ابی داؤد: رقم الحدیث 4710|لَا تُجَالِسُوْا=تم نہ بیٹھو~أَهْلَ الْقَدَرِ=قدریہ~لَا تُفَاتِحُوْهُمْ=ان سے سلام و کلام نہ کرو|حضرت عمر بن خطاب رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا کہ تم قدریہ (فرقہ کے لوگوں) کے ساتھ نہ بیٹھو اور نہ ہی ان سے سلام و کلام (کی ابتداء) کرو۔
# پانچویں فصل: مقام صحابہ کرام رضی اللہ عنہم
1|عَنْ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: إِنَّ اللّٰهَ عَزَّ وَجَلَّ نَظَرَ فِيْ قُلُوْبِ الْعِبَادِ فَاخْتَارَ مُحَمَّدًا فَبَعَثَهٗ بِرِسَالَاتِهٖ وَانْتَخَبَهٗ بِعِلْمِهٖ ثُمَّ نَظَرَ فِيْ قُلُوْبِ النَّاسِ بَعْدَهٗ فَاخْتَارَ لَهٗ أَصْحَابَهٗ فَجَعَلَهُمْ أَنْصَارَ دِيْنِهٖ وَوُزَرَاءَ نَبِيِّهٖ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ.|مسند ابی داؤد الطیالسی: رقم الحدیث 243|نَظَرَ=دیکھا~الْعِبَادِ=بندے~فَاخْتَارَ=پس اس نے منتخب کیا~مُحَمَّدًا=حضرت محمد صلی اللہ علیہ وسلم کو~فَبَعَثَهٗ=اس نے آپ کو مبعوث کیا~بِرِسَالَاتِهٖ=اپنے پیغامات دے کر~انْتَخَبَهٗ=اس نے آپ کو منتخب کیا~بِعِلْمِهٖ=اپنے علم کے ساتھ~لَهٗ=اس کے لیے~أَصْحَابَهٗ=اس کے صحابہ~فَجَعَلَهُمْ=پھر اس نے ان کو بنا دیا~أَنْصَارَ=مددگار~دِيْنِهٖ=اپنے دین~وُزَرَاءَ=وزیر|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے فرماتے ہیں کہ بے شک اللہ تعالیٰ نے بندوں کے دلوں کو دیکھا تو ان میں سے محمد صلی اللہ علیہ وسلم کو منتخب کیا، آپ کو اپنے پیغامات دے کر مبعوث کیا اور آپ کو اپنے علم کے ساتھ منتخب کیا۔ پھر اس کے بعد لوگوں کے دلوں کو دیکھا تو آپ صلی اللہ علیہ وسلم کے لیے اصحاب کو منتخب فرمایا، ان کو اپنے دین کا مددگار اور اپنے نبی صلی اللہ علیہ وسلم کا وزیر بنایا۔
2|عَنِ ابْنِ عَبَّاسٍ رَضِيَ اللّٰهُ عَنْهُمَا: ﴿سَلٰمٌ عَلٰى عِبَادِهِ الَّذِيْنَ اصْطَفٰى﴾ قَالَ: هُمْ أَصْحَابُ مُحَمَّدٍ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ اصْطَفَاهُمُ اللّٰهُ لِنَبِيِّهٖ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ.|مجمع الزوائد للہیثمی: رقم الحدیث 11249|سَلٰمٌ=سلامتی ہو~اصْطَفٰى=منتخب فرمایا~اصْطَفَاهُمُ=انہیں منتخب فرمایا~لِنَبِيِّهٖ=اپنے نبی صلی اللہ علیہ وسلم کے لیے|اللہ تعالیٰ کا فرمان ہے: ان لوگوں پر سلامتی ہو جنہیں اللہ نے منتخب فرمایا ہے۔ حضرت عبد اللہ بن عباس رضی اللہ عنہما فرماتے ہیں کہ اس آیت کا مصداق حضرت محمد صلی اللہ علیہ وسلم کے صحابہ ہیں، اللہ تعالیٰ نے انہیں اپنے نبی صلی اللہ علیہ وسلم کے لیے منتخب فرمایا ہے۔
3|عَنْ عَبْدِ اللّٰهِ بْنِ مُغَفَّلٍ الْمُزَنِيِّ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَللّٰهَ اللّٰهَ فِيْ أَصْحَابِيْ لَا تَتَّخِذُوْهُمْ غَرَضًا مِنْ بَعْدِيْ فَمَنْ أَحَبَّهُمْ فَبِحُبِّيْ أَحَبَّهُمْ وَمَنْ أَبْغَضَهُمْ فَبِبُغْضِيْ أَبْغَضَهُمْ وَمَنْ آذَاهُمْ فَقَدْ آذَانِيْ وَمَنْ آذَانِيْ فَقَدْ آذَى اللّٰهَ وَمَنْ آذَى اللّٰهَ يُوْشِكُ أَنْ يَأْخُذَهٗ.|شعب الایمان للبیہقی: رقم الحدیث 1511|اَللّٰهَ اللّٰهَ=اللہ سے ڈرو اللہ سے ڈرو~فِيْ أَصْحَابِيْ=میرے صحابہ کے بارے میں~لَا تَتَّخِذُوْهُمْ=تم ان کو نہ بناؤ~غَرَضًا=نشانہ~أَحَبَّهُمْ=ان سے محبت کرے~فَبِحُبِّيْ=میری محبت کی وجہ سے~أَبْغَضَهُمْ=ان سے بغض رکھتا ہے~فَبِبُغْضِيْ=مجھ سے بغض رکھنے کی وجہ سے~آذَاهُمْ=ان (صحابہ کرام) کو تکلیف پہنچائی~آذَانِيْ=اس نے مجھے تکلیف پہنچائی~آذَى اللّٰهَ=اس نے اللہ تعالیٰ کی نافرمانی کی~يُوْشِكُ=قریب ہے~أَنْ يَأْخُذَهٗ=کہ اس کو سزا دے گا|حضرت عبد اللہ بن مغفل مزنی رضی اللہ عنہ سے روایت ہے؛ فرماتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ سے میرے صحابہ کے بارے میں ڈرو۔ اللہ تعالیٰ سے میرے صحابہ کے بارے میں ڈرو۔ ان کو میرے بعد طعن و تشنیع کا نشانہ نہ بنانا۔ جو شخص صحابہ کرام سے محبت کرتا ہے وہ میری محبت کی وجہ سے ان سے محبت کرتا ہے اور جو شخص ان سے بغض رکھتا ہے وہ مجھ سے بغض رکھنے کی وجہ سے ان سے بغض رکھتا ہے اور جس شخص نے ان کو تکلیف پہنچائی تو اس نے مجھے تکلیف پہنچائی اور جس شخص نے مجھے تکلیف پہنچائی اس نے اللہ تعالیٰ کی نافرمانی کی اور جس شخص نے اللہ تعالیٰ کی نافرمانی کی تو اللہ تعالیٰ اس کو جلد پکڑ لے گا۔
# چھٹی فصل: مقام اہل بیت رضی اللہ عنہم
1|عَنِ ابْنِ عَبَّاسٍ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَحِبُّوا اللّٰهَ لِمَا يَغْذُوْكُمْ مِنْ نِعَمِهٖ وَأَحِبُّوْنِيْ بِحُبِّ اللّٰهِ وَأَحِبُّوْا أَهْلَ بَيْتِيْ بِحُبِّيْ.|سنن الترمذی: رقم الحدیث 3789|أَحِبُّوا اللّٰهَ=اللہ سے محبت کرو~لِمَا=اس وجہ سے~يَغْذُوْكُمْ=وہ تمہیں کھلاتا ہے~مِنْ نِعَمِهٖ=اپنی نعمتوں میں سے~أَحِبُّوْنِيْ=مجھ سے محبت کرو~بِحُبِّ اللّٰهِ=اللہ سے محبت کی وجہ سے~أَهْلَ بَيْتِيْ=میرے اہل بیت~بِحُبِّيْ=میری محبت کی وجہ سے|حضرت عبد اللہ بن عباس رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ سے محبت کرو ان نعمتوں کی وجہ سے جو وہ تمہیں کھلا رہا ہے اور اللہ کی محبت کی وجہ سے مجھ سے محبت کرو اور میری محبت کی وجہ سے میرے اہل بیت سے محبت کرو۔
2|عَنْ جَابِرِ بْنِ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: رَأَيْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ فِيْ حَجَّتِهٖ يَوْمَ عَرَفَةَ وَهُوَ عَلٰى نَاقَتِهِ الْقَصْوَاءِ يَخْطُبُ فَسَمِعْتُهٗ يَقُوْلُ: يَا أَيُّهَا النَّاسُ إِنِّيْ تَرَكْتُ فِيْكُمْ مَا إِنْ أَخَذْتُمْ بِهٖ لَنْ تَضِلُّوْا كِتَابَ اللّٰهِ وَعِتْرَتِيْ أَهْلَ بَيْتِيْ.|سنن الترمذی: رقم الحدیث 3786|رَأَيْتُ=میں نے دیکھا~رَسُوْلَ اللّٰهِ=رسول اللہ صلی اللہ علیہ وسلم~فِيْ حَجَّتِهٖ=حجۃ الوداع میں~يَوْمَ عَرَفَةَ=عرفہ کے دن~هُوَ=وہ~عَلٰى=پر~نَاقَتِهِ الْقَصْوَاءِ=اپنی قصویٰ نامی اونٹنی~يَخْطُبُ=خطبہ دے رہے تھے~فَسَمِعْتُهٗ=پس میں نے انہیں سنا~يَقُوْلُ=فرماتے ہوئے~يَا أَيُّهَا النَّاسُ=اے لوگو~إِنِّيْ=بے شک میں~تَرَكْتُ=چھوڑ کر جا رہا ہوں~فِيْكُمْ=تم میں~مَا=جو~إِنْ=اگر~أَخَذْتُمْ بِهٖ=تم اسے پکڑ رکھو گے~لَنْ تَضِلُّوْا=ہرگز گمراہ نہیں ہو گے~كِتَابَ اللّٰهِ=اللہ کی کتاب~عِتْرَتِيْ=میری اولاد~أَهْلَ بَيْتِيْ=میرے اہل بیت|حضرت جابر بن عبد اللہ رضی اللہ عنہما سے روایت ہے کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو حجۃ الوداع میں عرفہ کے دن اپنی قصواء نامی اونٹنی پر سوار ہو کر خطبہ دیتے ہوئے دیکھا اور یہ فرماتے ہوئے سنا: اے لوگو! میں تمہارے درمیان ایسی چیزیں چھوڑ کر جا رہا ہوں کہ اگر تم انہیں پکڑے رکھو گے تو ہرگز گمراہ نہ ہو گے۔ (پہلی چیز) اللہ تعالیٰ کی کتاب اور (دوسری چیز) میرے اہل بیت یعنی گھر والے ہیں۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَنْ سَرَّهٗ أَنْ يَكْتَالَ بِالْمِكْيَالِ الْأَوْفٰى إِذَا صَلّٰى عَلَيْنَا أَهْلَ الْبَيْتِ فَلْيَقُلْ: اَللّٰهُمَّ صَلِّ عَلٰى مُحَمَّدٍ وَأَزْوَاجِهٖ أُمَّهَاتِ الْمُؤْمِنِيْنَ وَذُرِّيَّتِهٖ وَأَهْلِ بَيْتِهٖ كَمَا صَلَّيْتَ عَلٰى آلِ إِبْرَاهِيْمَ إِنَّكَ حَمِيْدٌ مَجِيْدٌ.|سنن ابی داؤد: رقم الحدیث 982|مَنْ=جو شخص~سَرَّهٗ=اس کو خوش کرتی ہو~أَنْ يَكْتَالَ=یہ بات کہ تول کرے~بِالْمِكْيَالِ الْأَوْفٰى=مکمل تول کے ساتھ~إِذَا=جب~صَلّٰى=درود بھیجے~عَلَيْنَا أَهْلَ الْبَيْتِ=ہم اہل بیت پر~فَلْيَقُلْ=پس یوں کہے~اَللّٰهُمَّ=اے اللہ~صَلِّ=درود بھیج~عَلٰى مُحَمَّدٍ=حضرت محمد صلی اللہ علیہ وسلم پر~أَزْوَاجِهٖ=آپ کی ازواج مطہرات~أُمَّهَاتِ الْمُؤْمِنِيْنَ=امہات المؤمنین~ذُرِّيَّتِهٖ=آپ کی اولاد~أَهْلِ بَيْتِهٖ=آپ کے اہل بیت~كَمَا=جیسا کہ~صَلَّيْتَ=تو نے درود بھیجا~عَلٰى آلِ إِبْرَاهِيْمَ=آل ابراہیم پر~إِنَّكَ=بے شک تو~حَمِيْدٌ=قابل تعریف~مَجِيْدٌ=بزرگی والا|حضرت ابوہریرہ رضی اللہ عنہ نبی اکرم صلی اللہ علیہ وسلم سے روایت نقل کرتے ہیں کہ آپ صلی اللہ علیہ وسلم نے فرمایا: جس آدمی کو یہ بات خوش کرتی ہو کہ اس کے نامہ اعمال کا اسے پورے کا پورا بدلہ دیا جائے تو اسے چاہیے کہ جب وہ ہم اہل بیت پر درود بھیجے تو یوں کہا کرے: اللّٰہم صل علی محمد ... اے اللہ! حضرت محمد صلی اللہ علیہ وسلم اور آپ کی ازواج مطہرات؛ امہات المؤمنین پر اور آپ صلی اللہ علیہ وسلم کی اولاد پر اور آپ کے اہل بیت پر رحمتیں نازل فرما جیسا کہ تو نے آل ابراہیم پر رحمتیں نازل فرمائیں، بے شک تو قابل تعریف بزرگی والا ہے۔
# ساتویں فصل: حیات عیسیٰ علیہ السلام
1|أَنَّ سَعِيْدًا الْمَقْبُرِيَّ أَخْبَرَهٗ أَنَّهٗ سَمِعَ أَبَا هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ يَقُوْلُ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: وَالَّذِيْ نَفْسُ أَبِي الْقَاسِمِ بِيَدِهٖ لَيَنْزِلَنَّ عِيْسَى ابْنُ مَرْيَمَ إِمَامًا مُقْسِطًا وَحَكَمًا عَدْلًا فَلَيَكْسِرَنَّ الصَّلِيْبَ وَلَيَقْتُلَنَّ الْخِنْزِيْرَ وَلَيُصْلِحَنَّ ذَاتَ الْبَيْنِ وَلَيُذْهِبَنَّ الشَّحْنَاءَ وَلَيُعْرَضَنَّ عَلَيْهِ الْمَالُ فَلَا يَقْبَلُهٗ ثُمَّ لَئِنْ قَامَ عَلٰى قَبْرِيْ فَقَالَ: يَا مُحَمَّدُ لَأُجِيْبَنَّهٗ.|مسند ابی یعلیٰ الموصلی: ص 1149، رقم الحدیث 6577|الَّذِيْ=اس ذات کی قسم~نَفْسُ=جان~أَبِي الْقَاسِمِ=ابو القاسم~بِيَدِهٖ=اس کے قبضہ قدرت میں~لَيَنْزِلَنَّ=ضرور نازل ہوں گے~عِيْسَى ابْنُ مَرْيَمَ=عیسیٰ ابن مریم علیہما السلام~إِمَامًا=امام~مُقْسِطًا=انصاف پسند~حَكَمًا=حکمران~عَدْلًا=عادل~فَلَيَكْسِرَنَّ=ضرور توڑیں گے~الصَّلِيْبَ=صلیب~لَيَقْتُلَنَّ=قتل کریں گے~الْخِنْزِيْرَ=خنزیر~لَيُصْلِحَنَّ=ختم کریں گے~ذَاتَ الْبَيْنِ=باہمی عداوتوں~لَيُذْهِبَنَّ=ختم کر دیں گے~الشَّحْنَاءَ=آپس کا بغض~لَيُعْرَضَنَّ=پیش کیا جائے گا~عَلَيْهِ=ان پر~الْمَالُ=مال~فَلَا يَقْبَلُهٗ=وہ قبول نہیں کریں گے~ثُمَّ=پھر~لَئِنْ قَامَ=یقیناً کھڑے ہوں گے~عَلٰى قَبْرِيْ=میری قبر پر~فَقَالَ=پھر وہ کہیں گے~يَا مُحَمَّدُ=اے محمد~لَأُجِيْبَنَّهٗ=میں ضرور انہیں جواب دوں گا|حضرت سعید المقبری رحمہ اللہ سے روایت ہے؛ انہوں نے حضرت ابوہریرہ رضی اللہ عنہ کو یہ کہتے ہوئے سنا کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو یہ فرماتے ہوئے سنا: اس ذات کی قسم جس کے قبضہ قدرت میں ابو القاسم کی جان ہے عیسیٰ ابن مریم ایک عادل حکمران بن کر ضرور نازل ہوں گے، صلیب کو توڑیں گے، خنزیر کو قتل کریں گے، باہمی عداوتوں کو ختم کریں گے، آپس کا بغض ختم ہو جائے گا، ان پر مال پیش کیا جائے گا مگر وہ قبول نہیں کریں گے۔ پھر وہ میری قبر پر آئیں گے اور یا محمد! کہیں گے تو میں انہیں جواب دوں گا۔
2|عَنْ عَبْدِ اللّٰهِ بْنِ عَمْرٍو رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: يَنْزِلُ الْمَسِيْحُ ابْنُ مَرْيَمَ عَلَيْهِمَا السَّلَامُ فَإِذَا رَآهُ الدَّجَّالُ ذَابَ كَمَا تَذُوْبُ الشَّحْمَةُ قَالَ: فَيَقْتُلُ الدَّجَّالَ.|مصنف ابن ابی شیبہ: رقم الحدیث 38649|يَنْزِلُ=نازل ہوں گے~الْمَسِيْحُ ابْنُ مَرْيَمَ=عیسیٰ ابن مریم~إِذَا=جب~رَآهُ=ان کو دیکھے گا~الدَّجَّالُ=دجال~ذَابَ=پگھلنے لگے گا~كَمَا=جیسا کہ~تَذُوْبُ=پگھلتی ہے~الشَّحْمَةُ=چربی~فَيَقْتُلُ=پھر وہ قتل کر دیں گے~الدَّجَّالَ=دجال|حضرت عبد اللہ بن عمرو رضی اللہ عنہما فرماتے ہیں کہ عیسیٰ ابن مریم علیہ السلام نازل ہوں گے۔ جب دجال ان کو دیکھے گا تو (ڈر کی وجہ سے) ایسے پگھلنے لگے گا جیسے چربی پگھلتی ہے۔ فرمایا: پھر وہ دجال کو قتل کر دیں گے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: كَيْفَ أَنْتُمْ إِذَا نَزَلَ ابْنُ مَرْيَمَ مِنَ السَّمَاءِ فِيْكُمْ وَإِمَامُكُمْ مِنْكُمْ.|کتاب الاسماء والصفات للبیہقی: ج 2، ص 166|كَيْفَ=کیسا عالم ہوگا~أَنْتُمْ=تمہارا~إِذَا=جب~نَزَلَ=نازل ہوں گے~ابْنُ مَرْيَمَ=عیسیٰ ابن مریم~مِنَ السَّمَاءِ=آسمان سے~فِيْكُمْ=تم میں~إِمَامُكُمْ=تمہارا امام~مِنْكُمْ=تم میں سے|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے، فرماتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: تمہاری خوشی کا اس وقت کیا عالم ہوگا جب عیسیٰ ابن مریم [علیہ السلام] تم میں آسمان سے نازل ہوں گے اور تمہارا امام تم میں سے ہوگا۔
# آٹھویں فصل: ظہور مہدی علیہ الرضوان
1|عَنْ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَا تَذْهَبُ الدُّنْيَا حَتّٰى يَمْلِكَ الْعَرَبَ رَجُلٌ مِنْ أَهْلِ بَيْتِيْ يُوَاطِئُ اسْمُهُ اسْمِيْ.|سنن الترمذی: رقم الحدیث 2230|لَا تَذْهَبُ=ختم نہیں ہوگی~الدُّنْيَا=دنیا~حَتّٰى=یہاں تک کہ~يَمْلِكَ=حکومت کرے گا~الْعَرَبَ=عرب پر~رَجُلٌ=ایک شخص~مِنْ=سے~أَهْلِ بَيْتِيْ=میرے اہل بیت~يُوَاطِئُ=موافق ہوگا~اسْمُهٗ=اس شخص کا نام~اسْمِيْ=میرا نام|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: دنیا اس وقت تک ختم نہیں ہوگی جب تک کہ میرے اہل بیت سے ایک شخص عرب پر حکومت نہ کر لے، جس کا نام میرے نام کی طرح ہوگا۔
2|عَنْ سَعِيْدِ بْنِ الْمُسَيَّبِ رَحِمَهُ اللّٰهُ قَالَ: كُنَّا عِنْدَ أُمِّ سَلَمَةَ رَضِيَ اللّٰهُ عَنْهَا فَتَذَاكَرْنَا الْمَهْدِيَّ فَقَالَتْ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: اَلْمَهْدِيُّ مِنْ وَلَدِ فَاطِمَةَ.|سنن ابن ماجہ: رقم الحدیث 4086|كُنَّا=ہم تھے~عِنْدَ أُمِّ سَلَمَةَ=حضرت ام سلمہ رضی اللہ عنہا کی خدمت میں~فَتَذَاكَرْنَا=ہم نے تذکرہ کیا~الْمَهْدِيَّ=حضرت مہدی کا~فَقَالَتْ=آپ رضی اللہ عنہا فرمانے لگیں~سَمِعْتُ=میں نے سنا ہے~يَقُوْلُ=فرماتے ہوئے~الْمَهْدِيُّ=مہدی~مِنْ=سے~وَلَدِ فَاطِمَةَ=فاطمہ رضی اللہ عنہا کی اولاد|حضرت سعید بن مسیب رحمہ اللہ سے مروی ہے۔ فرماتے ہیں کہ ہم حضرت ام سلمہ رضی اللہ عنہا کی خدمت میں حاضر تھے کہ حضرت مہدی کا تذکرہ ہوا۔ آپ رضی اللہ عنہا فرمانے لگیں کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو یہ فرماتے ہوئے سنا ہے کہ مہدی (حضرت) فاطمہ (رضی اللہ عنہا) کی اولاد میں سے ہوں گے۔
3|عَنْ أُمِّ سَلَمَةَ زَوْجِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: يَكُوْنُ اخْتِلَافٌ عِنْدَ مَوْتِ خَلِيْفَةٍ فَيَخْرُجُ رَجُلٌ مِنْ أَهْلِ الْمَدِيْنَةِ هَارِبًا إِلٰى مَكَّةَ فَيَأْتِيْهِ نَاسٌ مِنْ أَهْلِ مَكَّةَ فَيُخْرِجُوْنَهٗ وَهُوَ كَارِهٌ فَيُبَايِعُوْنَهٗ بَيْنَ الرُّكْنِ وَالْمَقَامِ وَيُبْعَثُ إِلَيْهِ بَعْثٌ مِنَ الشَّامِ فَيُخْسَفُ بِهِمْ بِالْبَيْدَاءِ بَيْنَ مَكَّةَ وَالْمَدِيْنَةِ فَإِذَا رَأَى النَّاسُ ذٰلِكَ أَتَاهُ أَبْدَالُ الشَّامِ وَعَصَائِبُ أَهْلِ الْعِرَاقِ فَيُبَايِعُوْنَهٗ ثُمَّ يَنْشَأُ رَجُلٌ مِنْ قُرَيْشٍ أَخْوَالُهٗ كَلْبٌ فَيَبْعَثُ إِلَيْهِمْ بَعْثًا فَيَظْهَرُوْنَ عَلَيْهِمْ وَذٰلِكَ بَعْثُ كَلْبٍ وَالْخَيْبَةُ لِمَنْ لَمْ يَشْهَدْ غَنِيْمَةَ كَلْبٍ فَيَقْسِمُ الْمَالَ وَيَعْمَلُ فِي النَّاسِ بِسُنَّةِ نَبِيِّهِمْ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ وَيُلْقِي الْإِسْلَامُ بِجِرَانِهٖ إِلَى الْأَرْضِ فَيَلْبَثُ سَبْعَ سِنِيْنَ ثُمَّ يُتَوَفّٰى وَيُصَلِّيْ عَلَيْهِ الْمُسْلِمُوْنَ.|سنن ابی داؤد: رقم الحدیث 4286|يَكُوْنُ=ہوگا~اخْتِلَافٌ=اختلاف~عِنْدَ=وقت~مَوْتِ=موت~خَلِيْفَةٍ=خلیفہ~فَيَخْرُجُ=پس نکلے گا~رَجُلٌ=ایک آدمی~مِنْ أَهْلِ الْمَدِيْنَةِ=اہل مدینہ سے~هَارِبًا=بھاگ کر~إِلٰى مَكَّةَ=مکہ کی طرف~فَيَأْتِيْهِ=اس کے پاس آئیں گے~نَاسٌ=کچھ لوگ~فَيُخْرِجُوْنَهٗ=وہ اسے باہر نکالیں گے~هُوَ كَارِهٌ=اور وہ اس بات کو ناپسند کرنے والا ہوگا~فَيُبَايِعُوْنَهٗ=پس یہ لوگ ان کی بیعت کریں گے~بَيْنَ=درمیان~الرُّكْنِ=حجر اسود والا کونہ~الْمَقَامِ=مقام ابراہیم~يُبْعَثُ=بھیجا جائے گا~إِلَيْهِ=ان کی طرف~بَعْثٌ=لشکر~مِنَ الشَّامِ=شام کی طرف سے~فَيُخْسَفُ بِهِمْ=ان کو دھنسا دیا جائے گا~بِالْبَيْدَاءِ=مقام بیداء پر~فَإِذَا رَأَى=پس جب دیکھیں گے~النَّاسُ=لوگ~ذٰلِكَ=یہ (منظر)~أَتَاهُ=اس کے پاس آئیں گے~أَبْدَالُ الشَّامِ=شام کے ابدال~عَصَائِبُ=جماعتیں~أَهْلِ الْعِرَاقِ=عراق والوں کی~فَيُبَايِعُوْنَهٗ=پس وہ اس سے بیعت کریں گے~يَنْشَأُ=نکلے گا~قُرَيْشٍ=قریش~أَخْوَالُهٗ كَلْبٌ=اس کا ننھیالی قبیلہ بنو کلب ہوگا~فَيَبْعَثُ=وہ شخص بھیجے گا~إِلَيْهِمْ=ان کے خلاف~بَعْثًا=ایک لشکر~فَيَظْهَرُوْنَ=پس وہ (لشکر مہدی) غالب آ جائیں گے~عَلَيْهِمْ=ان پر~بَعْثُ كَلْبٍ=بنو کلب کا لشکر~الْخَيْبَةُ=ناکامی~لِمَنْ=اس شخص کے لیے~لَمْ يَشْهَدْ=جو حاضر نہ ہوا~غَنِيْمَةِ كَلْبٍ=بنو کلب کی غنیمت~فَيَقْسِمُ=پس وہ شخص (امام مہدی) تقسیم کرے گا~الْمَالَ=مال~يَعْمَلُ=عمل جاری کریں گے~فِي النَّاسِ=لوگوں میں~بِسُنَّةِ نَبِيِّهِمْ=ان کے نبی صلی اللہ علیہ وسلم کی سنت~يُلْقِي=ڈال دے گا~الْإِسْلَامُ=اسلام~بِجِرَانِهٖ=اپنی گردن~إِلَى الْأَرْضِ=زمین کی طرف~فَيَلْبَثُ=پس وہ رہیں گے~سَبْعَ سِنِيْنَ=سات سال~ثُمَّ يُتَوَفّٰى=پھر وفات پا جائیں گے~يُصَلِّيْ عَلَيْهِ=ان پر نماز جنازہ پڑھیں گے~الْمُسْلِمُوْنَ=مسلمان|حضرت ام سلمہ زوجہ رسول اللہ صلی اللہ علیہ وسلم سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: ایک خلیفہ کی وفات کے وقت امت میں اختلاف ہوگا (کہ کس کو خلیفہ بنائیں) تو ایک شخص مدینہ منورہ سے مکہ مکرمہ کی طرف بھاگ نکلے گا تو مکہ والوں میں سے کچھ لوگ ان کے پاس آئیں گے انہیں باہر لائیں گے (یعنی وہ کسی گھر میں ہوں گے تو لوگ وہاں سے ان کو باہر لائیں گے) حالانکہ یہ بات انہیں ناپسند ہوگی۔ یہ لوگ ان کے ہاتھ پر مقام ابراہیم اور حجر اسود کے درمیان بیعت کریں گے۔ (بیعت کے بعد) ان کی طرف شام سے ایک لشکر روانہ ہوگا لیکن اسے مکہ اور مدینہ کے درمیان مقام بیداء میں دھنسا دیا جائے گا۔ جب لوگ یہ منظر دیکھیں گے تو پھر ان کے پاس ابدال شام اور اہل عراق کی جماعتیں آنا شروع ہوں گی اور ان کے ہاتھ پر بیعت کریں گی۔ پھر قریش میں سے ایک شخص نکلے گا جس کا ننھیالی قبیلہ بنو کلب ہوگا، وہ شخص ان کی طرف ایک لشکر بھیجے گا تو یہ (یعنی امام مہدی کا لشکر) ان پر غالب آ جائیں گے۔ یہی بنی کلب کا لشکر ہوگا۔ ناکام رہے وہ شخص جو بنو کلب کے مال غنیمت میں حاضر نہ ہوا۔ وہ شخص (امام مہدی) مال غنیمت تقسیم کریں گے اور لوگوں میں ان کے نبی کی سنت کو جاری کریں گے۔ اس وقت اسلام اپنی گردن زمین میں ڈال دے گا (یعنی جس طرح اونٹ جب سکون سے بیٹھتا ہے تو اپنی گردن زمین پر بچھا دیتا ہے، اسی طرح یہ دور بھی اسلام کے لیے امن و سکون کا دور ہوگا) وہ (یعنی امام مہدی) سات سال تک حکمرانی کریں گے۔ پھر وفات پا جائیں گے اور مسلمان ان کی نماز جنازہ پڑھیں گے۔
# نویں فصل: قیامت
1|عَنْ أَنَسِ بْنِ مَالِكٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَثَلُ هٰذِهِ الدُّنْيَا مَثَلُ ثَوْبٍ شُقَّ مِنْ أَوَّلِهٖ إِلٰى آخِرِهٖ فَبَقِيَ مُتَعَلِّقًا بِخَيْطٍ فِيْ آخِرِهٖ فَيُوْشِكُ ذٰلِكَ الْخَيْطُ أَنْ يَنْقَطِعَ.|شعب الایمان للبیہقی: رقم الحدیث 10240|مَثَلُ=مثال~ثَوْبٍ=کپڑا~شُقَّ=پھاڑ دیا گیا~مِنْ=سے~أَوَّلِهٖ=اس کا شروع~إِلٰى=تک~آخِرِهٖ=اس کا آخر~فَبَقِيَ=پس باقی رہ گیا~مُتَعَلِّقًا=جڑا ہوا~بِخَيْطٍ=دھاگے کے ساتھ~فَيُوْشِكُ=پس قریب ہے~ذٰلِكَ=وہ~الْخَيْطُ=دھاگا~أَنْ يَنْقَطِعَ=کہ ٹوٹ جائے|حضرت انس رضی اللہ عنہ سے روایت ہے نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: اس دنیا کی مثال اس کپڑے کی طرح ہے جو اول سے آخر تک پھاڑ دیا گیا ہو اور آخری کنارے پر ایک دھاگے سے جڑا رہ گیا ہو اور وہ دھاگا کسی بھی وقت ٹوٹ سکتا ہے۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ أَنَّهٗ قَالَ: بُعِثْتُ أَنَا وَالسَّاعَةُ كَهَاتَيْنِ.|صحیح البخاری: رقم الحدیث 6504|بُعِثْتُ=مجھے بھیجا گیا~أَنَا=میں~السَّاعَةُ=قیامت~كَهَاتَيْنِ=ان دو کی طرح|حضرت انس رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا کہ مجھے اور قیامت کو یوں (ساتھ ساتھ) بھیجا گیا ہے جیسے یہ دو انگلیاں (ساتھ ساتھ) ہیں۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: يَعْرَقُ النَّاسُ يَوْمَ الْقِيَامَةِ حَتّٰى يَذْهَبَ عَرَقُهُمْ فِي الْأَرْضِ سَبْعِيْنَ ذِرَاعًا وَيُلْجِمُهُمْ حَتّٰى يَبْلُغَ آذَانَهُمْ.|صحیح البخاری: رقم الحدیث 6532|يَعْرَقُ=پسینہ آئے گا~حَتّٰى=یہاں تک~يَذْهَبَ=چلا جائے گا~عَرَقُهُمْ=ان کا پسینہ~الْأَرْضِ=زمین~سَبْعِيْنَ=ستر~ذِرَاعًا=ہاتھ~يُلْجِمُهُمْ=ان کو لگام دے گا~يَبْلُغَ=پہنچ جائے گا~آذَانَهُمْ=ان کے کانوں تک|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: قیامت کے دن لوگوں کو اتنا پسینہ آئے گا جو زمین کے اندر ستر ہاتھ تک چلا جائے گا اور پسینہ ان کے لیے لگام بن جائے گا یہاں تک کہ یہ پسینہ ان کے کانوں تک پہنچ جائے گا۔
# دسویں فصل: جنت و جہنم
1|عَنْ أَبِيْ سَعِيْدٍ الْخُدْرِيِّ وَأَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُمَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: يُنَادِيْ مُنَادٍ: إِنَّ لَكُمْ أَنْ تَصِحُّوْا فَلَا تَسْقَمُوْا أَبَدًا وَإِنَّ لَكُمْ أَنْ تَحْيَوْا فَلَا تَمُوْتُوْا أَبَدًا وَإِنَّ لَكُمْ أَنْ تَشِبُّوْا فَلَا تَهْرَمُوْا أَبَدًا وَإِنَّ لَكُمْ أَنْ تَنْعَمُوْا فَلَا تَبْأَسُوْا أَبَدًا.|صحیح مسلم: رقم الحدیث 7157|يُنَادِيْ=پکارے گا~مُنَادٍ=پکارنے والا~إِنَّ=بلاشبہ~لَكُمْ=تمہارے لیے~أَنْ تَصِحُّوْا=تم صحت مند رہو گے~فَلَا تَسْقَمُوْا=پس تم بیمار نہیں ہو گے~أَبَدًا=ہمیشہ~أَنْ تَحْيَوْا=تم زندہ رہو گے~فَلَا تَمُوْتُوْا=پس تم نہیں مرو گے~أَنْ تَشِبُّوْا=تم جوان رہو گے~فَلَا تَهْرَمُوْا=پس تم بوڑھے نہیں ہو گے~أَنْ تَنْعَمُوْا=تم ناز و نعمت میں رہو گے~فَلَا تَبْأَسُوْا=پس تمہیں تکلیف نہ ہوگی|حضرت ابو سعید خدری اور حضرت ابوہریرہ رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: (اللہ تعالیٰ کی طرف سے) ایک پکارنے والا (جنت میں جنتیوں کو مخاطب کر کے) پکارے گا کہ یہاں تم ہمیشہ صحت مند رہو گے اور کبھی بیمار نہ پڑو گے، تم ہمیشہ کے لیے زندہ رہو گے اور تم پر کبھی موت نہ آئے گی، تم ہمیشہ جوان رہو گے اور تم پر کبھی بڑھاپا نہیں آئے گا اور تم یہاں ناز و نعمت میں رہو گے اور تمہیں کبھی کوئی تنگی اور تکلیف نہ ہوگی۔
2|عَنْ أَبِيْ مُجَاهِدٍ حَدَّثَنَا أَبُوْ مُدِلَّةَ أَنَّهٗ سَمِعَ أَبَا هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ يَقُوْلُ: قُلْنَا: يَا رَسُوْلَ اللّٰهِ الْجَنَّةُ مَا بِنَاؤُهَا؟ قَالَ: لَبِنَةٌ مِنْ ذَهَبٍ وَلَبِنَةٌ مِنْ فِضَّةٍ مِلَاطُهَا الْمِسْكُ الْأَذْفَرُ وَحَصْبَاؤُهَا الْيَاقُوْتُ وَاللُّؤْلُؤُ وَتُرَابُهَا الزَّعْفَرَانُ مَنْ يَدْخُلُهَا يَخْلُدُ فِيْهَا يَنْعَمُ لَا يَبْؤُسُ لَا يَفْنٰى شَبَابُهُمْ وَلَا تَبْلٰى ثِيَابُهُمْ.|سنن الدارمی: رقم الحدیث 2821|بِنَاؤُهَا=اس کی تعمیر~لَبِنَةٌ=ایک اینٹ~مِنْ ذَهَبٍ=سونے کی~مِنْ فِضَّةٍ=چاندی کی~مِلَاطُهَا=اس کا گارا~الْمِسْكُ=مشک~الْأَذْفَرُ=نہایت تیز مہک والا~حَصْبَاؤُهَا=اس کے کنکر~اللُّؤْلُؤُ=موتی~الْيَاقُوْتُ=یاقوت~تُرَابُهَا=اس کی مٹی~الزَّعْفَرَانُ=زعفران~مَنْ يَدْخُلُهَا=جو شخص داخل ہوگا~يَخْلُدُ=وہ ہمیشہ رہے گا~فِيْهَا=اس میں~يَنْعَمُ=وہ خوش حال رہے گا~لَا يَبْؤُسُ=وہ بد حال نہیں ہوگا~لَا يَفْنٰى=ختم نہیں ہوگی~شَبَابُهُمْ=ان کی جوانی~لَا تَبْلٰى=پرانے نہیں ہوں گے~ثِيَابُهُمْ=ان کے کپڑے|حضرت مجاہد رحمہ اللہ سے مروی ہے وہ کہتے ہیں کہ ہمیں ابو مدلہ رحمہ اللہ نے بیان کیا، انہوں نے حضرت ابوہریرہ رضی اللہ عنہ سے سنا، وہ فرماتے ہیں کہ ہم نے پوچھا: یا رسول اللہ! جنت کی تعمیر کیسی ہے؟ آپ صلی اللہ علیہ وسلم نے فرمایا: اس کی تعمیر اس طرح ہے کہ ایک اینٹ سونے کی اور ایک اینٹ چاندی کی اور اس کا گارا تیز خوشبودار مشک ہے اور وہاں کے سنگریزے موتی اور یاقوت کے ہیں اور وہاں کی مٹی زعفران کی ہے۔ جو لوگ اس جنت میں داخل ہوں گے وہ ہمیشہ عیش اور سکون سے رہیں گے، کبھی بدحال نہیں ہوں گے (یعنی کوئی تنگی اور تکلیف ان کو نہ ہوگی) اور کبھی ان کے کپڑے پرانے اور خستہ نہ ہوں گے اور ان کی جوانی کبھی زائل نہیں ہوگی۔
3|عَنِ النُّعْمَانِ بْنِ بَشِيْرٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ أَهْوَنَ أَهْلِ النَّارِ عَذَابًا مَنْ لَهٗ نَعْلَانِ وَشِرَاكَانِ مِنْ نَارٍ يَغْلِيْ مِنْهُمَا دِمَاغُهٗ كَمَا يَغْلِي الْمِرْجَلُ مَا يَرٰى أَنَّ أَحَدًا أَشَدُّ مِنْهُ عَذَابًا وَإِنَّهٗ لَأَهْوَنُهُمْ عَذَابًا.|صحیح مسلم: رقم الحدیث 517|أَهْوَنَ=سب سے ہلکا~أَهْلِ النَّارِ=آگ والا~عَذَابًا=سزا~مَنْ=وہ شخص~لَهٗ=اس کے لیے~نَعْلَانِ=دو جوتے~شِرَاكَانِ=دو تسمے~مِنْ نَارٍ=آگ سے~يَغْلِيْ=کھولے گا~مِنْهُمَا=ان دونوں سے~دِمَاغُهٗ=اس کا دماغ~كَمَا=جس طرح~يَغْلِي=کھولتا ہے~الْمِرْجَلُ=ہنڈیا~مَا يَرٰى=وہ خیال نہیں کرے گا~أَحَدًا=کوئی ایک~أَشَدُّ=زیادہ سخت~مِنْهُ=اس سے~إِنَّهٗ=بلاشبہ وہ~لَأَهْوَنُهُمْ=یقیناً ان میں سب سے ہلکا|حضرت نعمان بن بشیر رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: دوزخیوں میں سب سے ہلکے عذاب والا وہ شخص ہوگا جس کے جوتے اور (ان کے) تسمے آگ کے ہوں گے۔ ان کی گرمی سے اس کا دماغ اس طرح کھولے گا جس طرح چولہے پر ہنڈیا کھولتی ہے۔ وہ یہ سمجھے گا کہ کوئی شخص اس سے زیادہ سخت عذاب میں نہیں ہے حالانکہ وہ (جہنمیوں میں) سب سے ہلکے عذاب والا ہوگا۔
4|عَنْ أَنَسِ بْنِ مَالِكٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: يَا أَيُّهَا النَّاسُ ابْكُوْا فَإِنْ لَمْ تَبْكُوْا فَتَبَاكَوْا فَإِنَّ أَهْلَ النَّارِ يَبْكُوْنَ فِي النَّارِ حَتّٰى تَسِيْلَ دُمُوْعُهُمْ فِيْ وُجُوْهِهِمْ كَأَنَّهَا جَدَاوِلُ حَتّٰى تَنْقَطِعَ الدُّمُوْعُ فَتَسِيْلَ الدِّمَاءُ فَتُقَرَّحَ الْعُيُوْنُ فَلَوْ أَنَّ سُفُنًا أُزْجِيَتْ فِيْهَا لَجَرَتْ.|مسند ابی یعلیٰ الموصلی: رقم الحدیث 4134|ابْكُوْا=تم روؤ~فَإِنْ لَمْ تَبْكُوْا=پس اگر تم رو نہیں سکتے~فَتَبَاكَوْا=تو رونے جیسی شکل بنا لو~أَهْلَ النَّارِ=آگ والے~يَبْكُوْنَ=وہ روئیں گے~فِي النَّارِ=آگ میں~حَتّٰى=یہاں تک~تَسِيْلَ=بہیں گے~دُمُوْعُهُمْ=ان کے آنسو~وُجُوْهِهِمْ=ان کے چہرے~كَأَنَّهَا=گویا کہ وہ~جَدَاوِلُ=چھوٹی نہریں~تَنْقَطِعَ=ختم ہو جائیں~الدُّمُوْعُ=آنسو~فَتَسِيْلَ=پس بہے گا~الدِّمَاءُ=خون~فَتُقَرَّحَ=پس وہ زخمی کر دے گا~الْعُيُوْنُ=آنکھیں~فَلَوْ=پس اگر~سُفُنًا=کشتیاں~أُزْجِيَتْ=چلائی جائیں~فِيْهَا=اس میں~لَجَرَتْ=البتہ چل سکیں|حضرت انس رضی اللہ عنہ سے روایت ہے کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو یہ فرماتے ہوئے سنا: اے لوگو! (اللہ کے عذاب کے خوف سے) خوب رویا کرو اور اگر تم رو نہ سکو (تمہیں رونا نہ آئے) تو پھر (کم از کم) رونے کی شکل بنا لو کیونکہ جہنمی دوزخ میں اتنا روئیں گے کہ ان کے چہروں پر ان کے آنسو ایسے بہیں گے گویا کہ وہ (بہتی ہوئی) نالیاں ہوں یہاں تک کہ آنسو ختم ہو جائیں گے اور پھر خون بہے گا اور پھر اس (خون بہنے) سے آنکھوں میں زخم پڑ جائیں گے اور (ان دوزخیوں کے ان آنسوؤں اور خون کی مجموعی مقدار اتنی ہو جائے گی کہ) اگر اس میں کشتیاں چلائی جائیں تو وہ بھی چل سکیں۔
## اَلْبَابُ الثَّانِيْ: اَلْعِبَادَاتُ — عبادات
# پہلی فصل: توحید کا اقرار
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أُمِرْتُ أَنْ أُقَاتِلَ النَّاسَ حَتّٰى يَقُوْلُوْا لَا إِلٰهَ إِلَّا اللّٰهُ فَمَنْ قَالَ لَا إِلٰهَ إِلَّا اللّٰهُ فَقَدْ عَصَمَ مِنِّيْ نَفْسَهٗ وَمَالَهٗ إِلَّا بِحَقِّهٖ وَحِسَابُهٗ عَلَى اللّٰهِ.|صحیح البخاری: رقم الحدیث 2946|أُمِرْتُ=مجھے حکم دیا گیا~أَنْ أُقَاتِلَ=کہ میں قتال کروں~يَقُوْلُوْا=وہ کہیں~عَصَمَ=محفوظ کر لیا~مِنِّيْ=مجھ سے~نَفْسَهٗ=اس کی جان~بِحَقِّهٖ=اس کے حق کی وجہ سے~حِسَابُهٗ=اس کا حساب~عَلَى اللّٰهِ=اللہ پر|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: مجھے حکم دیا گیا ہے کہ میں لوگوں سے اس وقت تک جہاد جاری رکھوں جب تک کہ وہ "لَا إِلٰهَ إِلَّا اللّٰهُ" نہ پڑھ لیں۔ جس شخص نے "لَا إِلٰهَ إِلَّا اللّٰهُ" پڑھ لیا اس نے اپنے مال اور اپنی جان کو محفوظ کر لیا۔ مگر (جان یا مال کے) حق کے بدلے میں (یعنی اگر اس نے کسی کی جان لی یا مال لیا تو اس کی سزا دنیا میں اسے ملے گی) اور اس کا حساب اللہ تعالیٰ کے سپرد ہے۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: يُخْرَجُ مِنَ النَّارِ مَنْ قَالَ: لَا إِلٰهَ إِلَّا اللّٰهُ وَكَانَ فِيْ قَلْبِهٖ مِنَ الْخَيْرِ مَا يَزِنُ شَعِيْرَةً ثُمَّ يُخْرَجُ مِنَ النَّارِ مَنْ قَالَ: لَا إِلٰهَ إِلَّا اللّٰهُ وَكَانَ فِيْ قَلْبِهٖ مِنَ الْخَيْرِ مَا يَزِنُ بُرَّةً ثُمَّ يُخْرَجُ مِنَ النَّارِ مَنْ قَالَ: لَا إِلٰهَ إِلَّا اللّٰهُ وَكَانَ فِيْ قَلْبِهٖ مِنَ الْخَيْرِ مَا يَزِنُ ذَرَّةً.|صحیح مسلم: رقم الحدیث 478|يُخْرَجُ=نکالا جائے گا~كَانَ=وہ تھا~فِيْ قَلْبِهٖ=اس کا دل~مِنَ الْخَيْرِ=بھلائی~مَا=وہ جو~يَزِنُ=برابر ہوگا~شَعِيْرَةً=جو~بُرَّةً=گیہوں کا دانہ~ذَرَّةً=ذرہ|حضرت انس رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جہنم سے وہ سب لوگ نکال دیے جائیں گے جنہوں نے "لَا إِلٰهَ إِلَّا اللّٰهُ" کہا اور ان کے دل میں جو کے دانے کے برابر خیر تھی۔ پھر وہ لوگ نکال دیے جائیں گے جنہوں نے "لَا إِلٰهَ إِلَّا اللّٰهُ" کہا اور ان کے دل میں گندم کے دانے کے برابر خیر تھی اور اس کے بعد وہ لوگ بھی نکال دیے جائیں گے جنہوں نے "لَا إِلٰهَ إِلَّا اللّٰهُ" کہا اور ان کے دل میں ذرہ برابر بھی خیر تھی۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَلْإِيْمَانُ بِضْعٌ وَسَبْعُوْنَ أَوْ بِضْعٌ وَسِتُّوْنَ شُعْبَةً فَأَفْضَلُهَا قَوْلُ لَا إِلٰهَ إِلَّا اللّٰهُ وَأَدْنَاهَا إِمَاطَةُ الْأَذٰى عَنِ الطَّرِيْقِ وَالْحَيَاءُ شُعْبَةٌ مِنَ الْإِيْمَانِ.|صحیح مسلم: رقم الحدیث 153|بِضْعٌ=کچھ~سَبْعُوْنَ=ستر~سِتُّوْنَ=ساٹھ~شُعْبَةً=شاخ~فَأَفْضَلُهَا=پس ان میں سب سے افضل~قَوْلُ=کہنا~أَدْنَاهَا=ان میں سب سے کم درجہ~إِمَاطَةُ=ہٹانا~الْأَذٰى=تکلیف دہ چیز~الطَّرِيْقِ=راستہ~الْحَيَاءُ=حیاء~شُعْبَةٌ=شاخ~الْإِيْمَانِ=ایمان|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: ایمان کی ساٹھ یا ستر سے کچھ زائد شاخیں ہیں اور ان میں سب سے اعلیٰ اور افضل "لَا إِلٰهَ إِلَّا اللّٰهُ" کا کہنا ہے اور ان میں ادنیٰ درجہ راستے سے تکلیف دینے والی چیزوں کا ہٹانا ہے اور حیا ایمان کی ایک اہم شاخ ہے۔
# دوسری فصل: نماز
1|عَنْ عُبَادَةَ بْنِ الصَّامِتِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: سَمِعْتُ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: خَمْسُ صَلَوَاتٍ افْتَرَضَهُنَّ اللّٰهُ عَلٰى عِبَادِهٖ فَمَنْ أَحْسَنَ وُضُوْءَهُنَّ وَصَلَّاهُنَّ لِوَقْتِهِنَّ وَأَتَمَّ رُكُوْعَهُنَّ وَسُجُوْدَهُنَّ وَخُشُوْعَهُنَّ كَانَ لَهٗ عَهْدٌ عَلَى اللّٰهِ أَنْ يَغْفِرَ لَهٗ وَمَنْ لَمْ يَفْعَلْ فَلَيْسَ لَهٗ عَلَى اللّٰهِ عَهْدٌ إِنْ شَاءَ عَذَّبَهٗ وَإِنْ شَاءَ غَفَرَ لَهٗ.|المعجم الاوسط للطبرانی: رقم الحدیث 4658|خَمْسُ=پانچ~افْتَرَضَهُنَّ=ان کو فرض کیا ہے~أَحْسَنَ=اچھا کیا~وُضُوْءَهُنَّ=ان کا وضو~صَلَّاهُنَّ=ان کو پڑھا~لِوَقْتِهِنَّ=ان کے وقت میں~أَتَمَّ=اس نے پورا کیا~رُكُوْعَهُنَّ=ان کا رکوع~سُجُوْدَهُنَّ=ان کا سجدہ~خُشُوْعَهُنَّ=ان کا خشوع~عَهْدٌ=ذمہ داری / وعدہ~يَغْفِرَ=وہ بخش دے گا~لَمْ يَفْعَلْ=نہیں کیا~عَذَّبَهٗ=اس کو عذاب دے|حضرت عبادہ بن صامت رضی اللہ عنہ سے روایت ہے، کہتے ہیں کہ میں نے نبی کریم صلی اللہ علیہ وسلم سے یہ فرماتے ہوئے سنا: اللہ تعالیٰ نے اپنے بندوں پر پانچ نمازیں فرض کی ہیں جس نے ان کی ادائیگی کے لیے اچھی طرح وضو کیا اور مقررہ وقت پر ان کو پڑھا اور رکوع سجود بھی صحیح طریقے سے کیا اور خشوع کے ساتھ ان کو ادا کیا تو ایسے شخص کے لیے اللہ تعالیٰ کا وعدہ ہے کہ وہ اس کو بخش دے گا اور جس نے ایسا نہیں کیا تو اس کے لیے اللہ تعالیٰ کا کوئی وعدہ نہیں ہے، اگر وہ چاہے تو اس کو سزا دے اور چاہے تو اس کو بخش دے۔
2|عَنْ أَبِيْ ذَرٍّ رَضِيَ اللّٰهُ عَنْهُ أَنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ خَرَجَ زَمَنَ الشِّتَاءِ وَالْوَرَقُ يَتَهَافَتُ فَأَخَذَ بِغُصْنَيْنِ مِنْ شَجَرَةٍ قَالَ: فَجَعَلَ ذٰلِكَ الْوَرَقُ يَتَهَافَتُ قَالَ فَقَالَ: يَا أَبَا ذَرٍّ قُلْتُ: لَبَّيْكَ يَا رَسُوْلَ اللّٰهِ! قَالَ: إِنَّ الْعَبْدَ الْمُسْلِمَ لَيُصَلِّي الصَّلَاةَ يُرِيْدُ بِهَا وَجْهَ اللّٰهِ فَتَهَافَتُ عَنْهُ ذُنُوْبُهٗ كَمَا يَتَهَافَتُ هٰذَا الْوَرَقُ عَنْ هٰذِهِ الشَّجَرَةِ.|مشکوٰۃ المصابیح: رقم الحدیث 576|خَرَجَ=باہر تشریف لائے~زَمَنَ=زمانہ~الشِّتَاءِ=سردی~الْوَرَقُ=پتے~يَتَهَافَتُ=جھڑ رہے تھے~فَأَخَذَ=انہوں نے پکڑا~بِغُصْنَيْنِ=دو ٹہنیوں کو~شَجَرَةٍ=درخت~فَجَعَلَ=پس شروع ہو گئے~ذٰلِكَ=وہ~لَبَّيْكَ=میں حاضر ہوں~لَيُصَلِّي=وہ پڑھتا ہے~يُرِيْدُ=وہ ارادہ کرتا ہے~وَجْهَ=رضامندی~ذُنُوْبُهٗ=اس کے گناہ|حضرت ابوذر غفاری رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم سردی کے موسم میں باہر تشریف لے گئے (خزاں کے موسم کی وجہ سے) درختوں کے پتے جھڑ رہے تھے۔ آپ صلی اللہ علیہ وسلم نے ایک درخت کی دو شاخیں پکڑیں۔ راوی فرماتے ہیں کہ جب آپ صلی اللہ علیہ وسلم نے شاخیں پکڑیں تو ان سے پتے اور زیادہ جھڑنے لگے۔ آپ صلی اللہ علیہ وسلم نے فرمایا: اے ابوذر! میں نے عرض کیا یا رسول اللہ! میں حاضر ہوں۔ آپ صلی اللہ علیہ وسلم نے فرمایا: جب ایک مومن خالصتاً اللہ تعالیٰ کے لیے نماز پڑھتا ہے تو اس کے (جسم سے) گناہ بھی ایسے ہی جھڑتے ہیں جس طرح اس درخت سے یہ پتے جھڑ رہے ہیں۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّهٗ سَمِعَ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: أَرَأَيْتُمْ لَوْ أَنَّ نَهَرًا بِبَابِ أَحَدِكُمْ يَغْتَسِلُ فِيْهِ كُلَّ يَوْمٍ خَمْسًا مَا تَقُوْلُ ذٰلِكَ يُبْقِيْ مِنْ دَرَنِهٖ؟ قَالُوْا: لَا يُبْقِيْ مِنْ دَرَنِهٖ شَيْئًا قَالَ: فَذٰلِكَ مَثَلُ الصَّلَوَاتِ الْخَمْسِ يَمْحُو اللّٰهُ بِهِ الْخَطَايَا.|صحیح البخاری: رقم الحدیث 528|أَرَأَيْتُمْ=تمہارا کیا خیال ہے~لَوْ=اگر~أَنَّ=تحقیق~نَهَرًا=نہر~بِبَابِ=دروازے کے ساتھ~أَحَدِكُمْ=تم میں سے کوئی~يَغْتَسِلُ=وہ غسل کرتا ہے~فِيْهِ=اس میں~كُلَّ يَوْمٍ=ہر دن~خَمْسًا=پانچ~مَا تَقُوْلُ=تو کیا کہتا ہے~يُبْقِيْ=باقی رکھے گا~مِنْ دَرَنِهٖ=اس کا میل کچیل~شَيْئًا=کوئی چیز~مَثَلُ=مثال~الصَّلَوَاتِ=نمازیں~الْخَمْسِ=پانچ~يَمْحُو=وہ مٹا دیتا ہے~بِهٖ=اس کی وجہ سے~الْخَطَايَا=گناہ|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ انہوں نے رسول اللہ صلی اللہ علیہ وسلم کو فرماتے ہوئے سنا: تم بتاؤ کہ جس آدمی کے دروازے کے آگے پانی کی نہر چلتی ہو اور وہ روزانہ اس میں پانچ مرتبہ نہاتا ہو تو کیا اس کے بدن پر کچھ میل کچیل باقی رہ جائے گا؟ صحابہ کرام رضی اللہ عنہم نے عرض کیا: نہیں! میل بالکل باقی نہیں بچے گا۔ آپ صلی اللہ علیہ وسلم نے فرمایا: پانچوں نمازوں کی مثال ایسے ہی ہے کہ اللہ تعالیٰ تمام (صغیرہ) گناہوں کو ان نمازوں کے سبب سے مٹا دیتا ہے۔
# تیسری فصل: زکوٰۃ
1|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ الصَّدَقَةَ لَتُطْفِئُ غَضَبَ الرَّبِّ وَتَدْفَعُ مِيْتَةَ السُّوْءِ.|سنن الترمذی: رقم الحدیث 664|لَتُطْفِئُ=البتہ بجھا دیتا ہے~غَضَبَ=ناراضی / غصہ~الرَّبِّ=پروردگار~تَدْفَعُ=دور کرتا ہے~مِيْتَةَ=موت~السُّوْءِ=بری|حضرت انس رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: صدقہ اللہ تعالیٰ کے غصہ کو ٹھنڈا کرتا ہے اور بری موت سے بچاتا ہے۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَا نَقَصَتْ صَدَقَةٌ مِنْ مَالٍ وَمَا زَادَ اللّٰهُ رَجُلًا بِعَفْوٍ إِلَّا عِزًّا وَمَا تَوَاضَعَ أَحَدٌ لِلّٰهِ إِلَّا رَفَعَهُ اللّٰهُ.|سنن الترمذی: رقم الحدیث 2029|مَا نَقَصَتْ=کم نہیں کرتا~مَا زَادَ=نہیں بڑھاتا~عِزًّا=عزت~مَا تَوَاضَعَ=عاجزی اختیار نہیں کرتا~رَفَعَهُ=اس کو بلند کر دیا|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: صدقہ سے مال میں کمی نہیں آتی اور معاف کر دینے سے اللہ تعالیٰ آدمی کی عزت میں اضافہ ہی فرماتے ہیں اور جو بندہ اللہ کے لیے عاجزی اور انکساری اختیار کرے؛ اللہ تعالیٰ اس کو رفعت اور بلندی بخشیں گے۔
3|عَنْ مَرْثَدِ بْنِ عَبْدِ اللّٰهِ الْيَزَنِيِّ رَحِمَهُ اللّٰهُ حَدَّثَنِيْ بَعْضُ أَصْحَابِ رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ أَنَّهٗ سَمِعَ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: إِنَّ ظِلَّ الْمُؤْمِنِ يَوْمَ الْقِيَامَةِ صَدَقَتُهٗ.|مسند احمد: رقم الحدیث 17966|إِنَّ=بلاشبہ~ظِلَّ=سایہ~الْمُؤْمِنِ=ایمان والا~يَوْمَ الْقِيَامَةِ=قیامت کے دن~صَدَقَتُهٗ=اس کا صدقہ|حضرت مرثد بن عبد اللہ رحمہ اللہ بیان کرتے ہیں کہ مجھ سے رسول اللہ صلی اللہ علیہ وسلم کے کسی صحابی نے بیان کیا کہ انہوں نے رسول اللہ صلی اللہ علیہ وسلم سے یہ بات سنی ہے کہ قیامت کے دن مومن پر اس کے صدقہ کا سایہ ہوگا۔
# چوتھی فصل: روزہ
1|عَنْ أَبِيْ أُمَامَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قُلْتُ يَا رَسُوْلَ اللّٰهِ! مُرْنِيْ بِأَمْرٍ يَنْفَعُنِيَ اللّٰهُ بِهٖ قَالَ: عَلَيْكَ بِالصِّيَامِ فَإِنَّهٗ لَا مِثْلَ لَهٗ.|سنن النسائی: رقم الحدیث 2223|مُرْنِيْ=آپ مجھے حکم دیں~بِأَمْرٍ=کسی کام کا~يَنْفَعُنِيْ=جو مجھے فائدہ دے~بِهٖ=اس کے ذریعے~عَلَيْكَ=لازم پکڑو~بِالصِّيَامِ=روزوں کو~فَإِنَّهٗ=بلاشبہ وہ~لَا=نہیں~مِثْلَ=مثال~لَهٗ=اس کی|حضرت ابو امامہ رضی اللہ عنہ سے روایت ہے کہ میں نے عرض کیا: اے اللہ کے رسول! مجھے کسی عمل کا حکم فرمائیے جس سے اللہ تعالیٰ مجھے نفع دے۔ آپ صلی اللہ علیہ وسلم نے ارشاد فرمایا: روزہ رکھا کرو کیونکہ اس کی مثل کوئی بھی عمل نہیں ہے۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ أَفْطَرَ يَوْمًا مِنْ رَمَضَانَ مِنْ غَيْرِ رُخْصَةٍ وَلَا مَرَضٍ لَمْ يَقْضِ عَنْهُ صَوْمُ الدَّهْرِ كُلِّهٖ وَإِنْ صَامَهٗ.|سنن الترمذی: رقم الحدیث 723|أَفْطَرَ=روزہ نہ رکھا~يَوْمًا=ایک دن~غَيْرِ=بغیر~رُخْصَةٍ=رخصت~مَرَضٍ=بیماری~لَمْ يَقْضِ=بدل نہیں ہو سکتا~عَنْهُ=اس سے~صَوْمُ=روزہ رکھنا~الدَّهْرِ=زمانہ~كُلِّهٖ=اس کا مکمل~وَإِنْ=اگرچہ~صَامَهٗ=وہ روزے رکھے|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جس آدمی نے شرعی رخصت یا بیماری جیسے کسی عذر کے بغیر رمضان کا ایک روزہ بھی چھوڑ دیا تو اگر وہ آدمی اس (رمضان) کے بدلے عمر بھر بھی روزے رکھتا رہے تو اس کا نعم البدل نہیں ہو سکتا۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَنْ صَامَ رَمَضَانَ إِيْمَانًا وَاحْتِسَابًا غُفِرَ لَهٗ مَا تَقَدَّمَ مِنْ ذَنْبِهٖ وَمَنْ قَامَ شَهْرَ رَمَضَانَ إِيْمَانًا وَاحْتِسَابًا غُفِرَ لَهٗ مَا تَقَدَّمَ مِنْ ذَنْبِهٖ وَمَنْ قَامَ لَيْلَةَ الْقَدْرِ إِيْمَانًا وَاحْتِسَابًا غُفِرَ لَهٗ مَا تَقَدَّمَ مِنْ ذَنْبِهٖ.|سنن النسائی: رقم الحدیث 2204|إِيْمَانًا=ایمان~احْتِسَابًا=ثواب کی نیت سے~غُفِرَ=بخش دیا گیا~تَقَدَّمَ=اس نے پہلے کیا~ذَنْبِهٖ=اس کا گناہ~قَامَ=اس نے قیام کیا~شَهْرَ رَمَضَانَ=رمضان کا مہینہ~لَيْلَةَ الْقَدْرِ=شب قدر|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا کہ جس نے رمضان کے روزے ایمان کے ساتھ اور ثواب کی نیت سے رکھے، اس کے گزشتہ گناہ معاف کر دیے جائیں گے اور جس نے ایمان کے ساتھ اور ثواب کی نیت سے ماہ رمضان کی راتوں میں قیام کیا (یعنی تراویح پڑھی) اس کے بھی پچھلے گناہ معاف کر دیے جائیں گے اور جس نے شب قدر میں ایمان کے ساتھ اور ثواب کی نیت سے قیام کیا اس کے بھی پہلے گناہ معاف کر دیے جائیں گے۔
# پانچویں فصل: حج
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ خَرَجَ حَاجًّا أَوْ مُعْتَمِرًا أَوْ غَازِيًا ثُمَّ مَاتَ فِيْ طَرِيْقِهٖ كَتَبَ اللّٰهُ لَهٗ أَجْرَ الْغَازِيْ وَالْحَاجِّ وَالْمُعْتَمِرِ إِلٰى يَوْمِ الْقِيَامَةِ.|شعب الایمان للبیہقی: رقم الحدیث 4100|خَرَجَ=نکلا~حَاجًّا=حج کرنے والا~مُعْتَمِرًا=عمرہ کرنے والا~غَازِيًا=جہاد کے لیے نکلنے والا~ثُمَّ=پھر~مَاتَ=وہ مر گیا~فِيْ طَرِيْقِهٖ=اس کا راستہ~كَتَبَ=اس نے لکھ دیا~أَجْرَ=ثواب / بدلہ~الْغَازِيْ=جہاد سے واپس لوٹنے والا~الْحَاجِّ=حج کرنے والا~الْمُعْتَمِرِ=عمرہ کرنے والا~إِلٰى=تک~يَوْمِ الْقِيَامَةِ=قیامت|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جو آدمی حج یا عمرہ کی نیت سے یا اللہ تعالیٰ کے راستے میں جہاد کے لیے نکلا پھر راستہ ہی میں اس کو موت آ گئی تو اللہ تعالیٰ کی طرف سے اس کے لیے قیامت تک وہی اجر و ثواب لکھا جاتا رہے گا جو اللہ تعالیٰ کے راستے میں جہاد کرنے والے اور حج و عمرہ کرنے کے لیے مقرر ہے۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: سَمِعْتُ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: مَنْ حَجَّ لِلّٰهِ فَلَمْ يَرْفُثْ وَلَمْ يَفْسُقْ رَجَعَ كَيَوْمِ وَلَدَتْهُ أُمُّهٗ.|صحیح البخاری: رقم الحدیث 1521|سَمِعْتُ=میں نے سنا~حَجَّ=اس نے حج کیا~لِلّٰهِ=اللہ کے لیے~فَلَمْ يَرْفُثْ=اس نے فحش بات نہیں کی~لَمْ يَفْسُقْ=اس نے گناہ نہیں کیا~رَجَعَ=وہ لوٹا~كَيَوْمِ=اس دن کی طرح~وَلَدَتْهُ=اس کو اس نے جنا~أُمُّهٗ=اس کی ماں|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ میں نے نبی اکرم صلی اللہ علیہ وسلم سے سنا، آپ صلی اللہ علیہ وسلم نے فرمایا: جس آدمی نے اللہ تعالیٰ کی رضا کے لیے حج کیا اور اس (حج) میں نہ تو کسی شہوت انگیز اور فحش بات کا ارتکاب کیا اور نہ ہی اللہ تعالیٰ کی کوئی نافرمانی کی تو وہ (گناہوں سے ایسا پاک و صاف ہو کر) واپس لوٹے گا جیسا کہ اس دن تھا جس دن اس کی ماں نے اسے جنا تھا۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنْ رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ أَنَّهٗ قَالَ: اَلْحَاجُّ وَالْعُمَّارُ وَفْدُ اللّٰهِ إِنْ دَعَوْهُ أَجَابَهُمْ وَإِنِ اسْتَغْفَرُوْهُ غَفَرَ لَهُمْ.|سنن ابن ماجہ: رقم الحدیث 2892|الْحَاجُّ=حج کرنے والے~الْعُمَّارُ=عمرہ کرنے والے~وَفْدُ اللّٰهِ=اللہ کے مہمان~إِنْ=اگر~دَعَوْهُ=وہ دعا کریں~أَجَابَهُمْ=ان کی دعا قبول کرتا ہے~اسْتَغْفَرُوْهُ=انہوں نے اس سے معافی چاہی~غَفَرَ=اس نے بخش دیا~لَهُمْ=ان کو|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: حج اور عمرہ کرنے والے اللہ تعالیٰ کے مہمان ہیں، اگر وہ اللہ سے دعا کریں تو اللہ تعالیٰ ان کی دعا کو قبول فرمائیں گے اور اگر وہ اس سے مغفرت مانگیں تو وہ ان کی مغفرت فرمائے گا۔
## اَلْبَابُ الثَّالِثُ: اَلْأَخْلَاقِيَّاتُ — اخلاقیات
# پہلی فصل: خوش خلقی
1|عَنْ عَائِشَةَ رَضِيَ اللّٰهُ عَنْهَا قَالَتْ: سَمِعْتُ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: إِنَّ الْمُؤْمِنَ يُدْرِكُ بِحُسْنِ خُلُقِهٖ دَرَجَاتِ قَائِمِ اللَّيْلِ صَائِمِ النَّهَارِ.|مسند احمد: رقم الحدیث 24236|يُدْرِكُ=پا لیتا ہے~بِحُسْنِ خُلُقِهٖ=اس کے اچھے اخلاق~دَرَجَاتِ=مرتبے~قَائِمِ=کھڑا ہونے والا~اللَّيْلِ=رات~صَائِمِ=روزہ دار~النَّهَارِ=دن|حضرت عائشہ صدیقہ رضی اللہ عنہا سے روایت ہے فرماتی ہیں کہ میں نے نبی اکرم صلی اللہ علیہ وسلم سے سنا: آپ فرماتے ہیں کہ مومن آدمی اچھے اخلاق کی وجہ سے اُن لوگوں کا درجہ حاصل کر لیتا ہے جو رات بھر نفلی نمازیں پڑھتے ہوں اور دن کو روزہ رکھتے ہوں۔
2|عَنْ زَيْدِ بْنِ أَسْلَمَ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّمَا بُعِثْتُ لِأُتَمِّمَ صَالِحَ الْأَخْلَاقِ.|مصنف ابن ابی شیبہ: رقم الحدیث 32433|بُعِثْتُ=مجھے بھیجا گیا~لِأُتَمِّمَ=تاکہ میں تکمیل کروں~صَالِحَ الْأَخْلَاقِ=اچھے اخلاق / اچھی صفات و عادات|حضرت زید بن اسلم رحمہ اللہ سے (مرسلاً) روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: میں اس لیے نبی بنا کر بھیجا گیا ہوں تاکہ اچھے اخلاق کی تکمیل کروں۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَكْمَلُ الْمُؤْمِنِيْنَ إِيْمَانًا أَحْسَنُهُمْ خُلُقًا.|صحیح ابن حبان: رقم الحدیث 479|أَكْمَلُ=سب سے زیادہ کامل~الْمُؤْمِنِيْنَ=ایمان لانے والے~أَحْسَنُهُمْ=ان میں سب سے اچھا~خُلُقًا=عادت|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: اہل ایمان میں ایمان کے لحاظ سے زیادہ کامل لوگ وہ ہیں جو ان میں سے اخلاق کے اعتبار سے سب سے زیادہ اچھے ہیں۔
# دوسری فصل: سخاوت
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَا يَجْتَمِعُ الشُّحُّ وَالْإِيْمَانُ فِيْ قَلْبِ عَبْدٍ أَبَدًا.|کنز العمال فی سنن الاقوال والافعال: رقم الحدیث 4710|لَا يَجْتَمِعُ=جمع نہیں ہو سکتے~الشُّحُّ=بخل~الْإِيْمَانُ=ایمان~فِيْ=میں~قَلْبِ=دل~عَبْدٍ=بندہ~أَبَدًا=ہمیشہ|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: بخل اور ایمان کبھی بھی ایک آدمی کے دل میں جمع نہیں ہو سکتے۔
2|عَنْ أَبِيْ بَكْرٍ الصِّدِّيْقِ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَا يَدْخُلُ الْجَنَّةَ خَبٌّ وَلَا بَخِيْلٌ وَلَا مَنَّانٌ.|سنن الترمذی: رقم الحدیث 1963|لَا يَدْخُلُ=وہ داخل نہیں ہوگا~الْجَنَّةَ=جنت~خَبٌّ=دھوکہ باز~بَخِيْلٌ=بخل کرنے والا~مَنَّانٌ=احسان جتلانے والا|حضرت ابو بکر صدیق رضی اللہ عنہ نبی اکرم صلی اللہ علیہ وسلم سے نقل کرتے ہیں کہ آپ نے ارشاد فرمایا: دھوکہ باز، بخیل اور احسان جتانے والا شخص (ابتداء) جنت میں داخل نہیں ہوگا۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: قَالَ اللّٰهُ: أَنْفِقْ أُنْفِقْ عَلَيْكَ.|صحیح البخاری: رقم الحدیث 4684|قَالَ=فرمایا~اللّٰهُ=اللہ~أَنْفِقْ=تو خرچ کر~أُنْفِقْ=میں خرچ کروں گا~عَلَيْكَ=تجھ پر|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ کا اپنے بندوں کو ارشاد ہے کہ تم (دوسروں پر) خرچ کرتے رہو میں تمہیں عطا کرتا رہوں گا۔
# تیسری فصل: حیا
1|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ لِكُلِّ دِيْنٍ خُلُقًا وَخُلُقُ الْإِسْلَامِ الْحَيَاءُ.|سنن ابن ماجہ: رقم الحدیث 4181|لِكُلِّ=ہر ایک~دِيْنٍ=دین~خُلُقًا=وصف / عادت~خُلُقُ=وصف / عادت~الْإِسْلَامِ=اسلام~الْحَيَاءُ=حیا|حضرت انس رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: ہر دین کا کوئی نہ کوئی امتیازی وصف ہوتا ہے اور دین اسلام کا امتیازی وصف حیا ہے۔
2|عَنْ سَعِيْدِ بْنِ جُبَيْرٍ قَالَ: قَالَ ابْنُ عُمَرَ رَضِيَ اللّٰهُ عَنْهُ: إِنَّ الْحَيَاءَ وَالْإِيْمَانَ قُرِنَا جَمِيْعًا فَإِذَا رُفِعَ أَحَدُهُمَا رُفِعَ الْآخَرُ.|مصنف ابن ابی شیبہ: رقم الحدیث 25859|قُرِنَا=ان دونوں کو ملا دیا گیا~جَمِيْعًا=اکٹھا~رُفِعَ=اٹھا لیا جائے~أَحَدُهُمَا=ان دونوں میں سے ایک~الْآخَرُ=دوسرا|حضرت سعید بن جبیر رحمہ اللہ سے مروی ہے وہ کہتے ہیں کہ حضرت عبد اللہ بن عمر رضی اللہ تعالیٰ عنہما نے ارشاد فرمایا: حیاء اور ایمان یہ دونوں ہمیشہ اکٹھے رہتے ہیں۔ جب ان دونوں (حیاء اور ایمان) میں سے کوئی ایک اُٹھا لیا جائے تو دوسرا بھی اٹھا لیا جاتا ہے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَلْحَيَاءُ مِنَ الْإِيْمَانِ وَالْإِيْمَانُ فِي الْجَنَّةِ وَالْبَذَاءُ مِنَ الْجَفَاءِ وَالْجَفَاءُ فِي النَّارِ.|سنن الترمذی: رقم الحدیث 2009|الْحَيَاءُ=حیاء~مِنَ=سے~الْإِيْمَانِ=ایمان~فِي الْجَنَّةِ=جنت میں~الْبَذَاءُ=فحش گو / بدکلام~الْجَفَاءِ=بدی~فِي النَّارِ=آگ میں|حضرت ابوہریرہ رضی اللہ تعالیٰ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: حیاء ایمان کی ایک شاخ ہے اور ایمان جنت کی طرف لے جاتا ہے اور بے حیائی برائی کی ایک شاخ ہے اور برائی جہنم کی طرف لے جانے والی ہے۔
# چوتھی فصل: تواضع
1|عَنْ عِيَاضِ بْنِ حِمَارٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّهٗ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ اللّٰهَ أَوْحٰى إِلَيَّ أَنْ تَوَاضَعُوْا حَتّٰى لَا يَبْغِيَ أَحَدٌ عَلٰى أَحَدٍ وَلَا يَفْخَرَ أَحَدٌ عَلٰى أَحَدٍ.|سنن ابی داؤد: رقم الحدیث 4895|أَوْحٰى=وحی کی~إِلَيَّ=میری طرف~أَنْ=یہ کہ~تَوَاضَعُوْا=تم تواضع اختیار کرو~حَتّٰى=یہاں تک~لَا يَبْغِيَ=ظلم نہ کرے~أَحَدٌ=کوئی~عَلٰى أَحَدٍ=کسی ایک پر~لَا يَفْخَرَ=فخر نہ کرے|حضرت عیاض بن حمار رضی اللہ تعالیٰ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ نے میری طرف وحی فرمائی ہے کہ تواضع اختیار کرو، اس طور پر کہ کوئی کسی پر ظلم و زیادتی نہ کرے اور نہ ہی کوئی کسی کے مقابلہ میں فخر کرے۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: ثَلَاثَةٌ لَا يُكَلِّمُهُمُ اللّٰهُ يَوْمَ الْقِيَامَةِ وَلَا يُزَكِّيْهِمْ قَالَ أَبُوْ مُعَاوِيَةَ: وَلَا يَنْظُرُ إِلَيْهِمْ وَلَهُمْ عَذَابٌ أَلِيْمٌ شَيْخٌ زَانٍ وَمَلِكٌ كَذَّابٌ وَعَائِلٌ مُسْتَكْبِرٌ.|صحیح مسلم: رقم الحدیث 296|ثَلَاثَةٌ=تین~لَا يُكَلِّمُهُمُ=ان سے کلام نہیں کرے گا~يَوْمَ الْقِيَامَةِ=قیامت کے دن~لَا يُزَكِّيْهِمْ=وہ ان کا تزکیہ نہیں کرے گا~لَا يَنْظُرُ=وہ نہیں دیکھے گا~إِلَيْهِمْ=ان کی طرف~لَهُمْ=ان کے لیے~عَذَابٌ=سزا~أَلِيْمٌ=دردناک~شَيْخٌ=بوڑھا~زَانٍ=بدکاری کرنے والا~مَلِكٌ=بادشاہ~كَذَّابٌ=بہت جھوٹ بولنے والا~عَائِلٌ=تنگدست~مُسْتَكْبِرٌ=بڑا پن دکھانے والا|حضرت ابوہریرہ رضی اللہ تعالیٰ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: تین آدمی ایسے ہیں جن سے اللہ تعالیٰ قیامت کے دن کلام نہیں فرمائے گا اور ان کو پاک نہیں کرے گا۔ حضرت ابو معاویہ رحمہ اللہ فرماتے ہیں: اللہ تعالیٰ ان کی طرف نظر رحمت بھی نہیں کرے گا اور ان کے لیے آخرت میں دردناک عذاب ہے، ایک بوڑھا زانی، دوسرا جھوٹا حکمران اور تیسرا غریب متکبر۔
3|عَنْ عَبْدِ اللّٰهِ بْنِ مَسْعُوْدٍ رَضِيَ اللّٰهُ عَنْهُ عَنْ رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَا يَدْخُلُ الْجَنَّةَ مَنْ كَانَ فِيْ قَلْبِهٖ مِثْقَالُ ذَرَّةٍ مِنْ كِبْرٍ.|صحیح مسلم: رقم الحدیث 267|لَا يَدْخُلُ=وہ داخل نہیں ہوگا~الْجَنَّةَ=جنت~فِيْ قَلْبِهٖ=اس کا دل~مِثْقَالُ=برابر~ذَرَّةٍ=ذرہ~كِبْرٍ=بڑائی / تکبر|حضرت عبد اللہ بن مسعود رضی اللہ تعالیٰ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جس (آدمی) کے دل میں ذرہ برابر بھی تکبر ہوگا وہ (ابتداء) جنت میں نہیں جائے گا۔
# پانچویں فصل: صبر
1|عَنِ ابْنِ عَبَّاسٍ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ أُصِيْبَ بِمُصِيْبَةٍ بِمَالِهٖ أَوْ فِيْ نَفْسِهٖ فَكَتَمَهَا وَلَمْ يَشْكُهَا إِلَى النَّاسِ كَانَ حَقًّا عَلَى اللّٰهِ أَنْ يَغْفِرَ لَهٗ.|المعجم الاوسط للطبرانی: رقم الحدیث 737|أُصِيْبَ=اسے پہنچایا گیا~بِمُصِيْبَةٍ=کسی مصیبت کے ساتھ~بِمَالِهٖ=اس کا مال~أَوْ=یا~فِيْ نَفْسِهٖ=اس کی ذات~فَكَتَمَهَا=اس نے اسے چھپایا~لَمْ يَشْكُهَا=اس کو ظاہر نہ کرے~إِلَى النَّاسِ=لوگوں تک / طرف~كَانَ=ہوا~حَقًّا=ذمہ~عَلَى اللّٰهِ=اللہ پر~أَنْ يَغْفِرَ=کہ وہ معاف کرے~لَهٗ=اس کے لیے|حضرت عبد اللہ بن عباس رضی اللہ تعالیٰ عنہما رسول اللہ صلی اللہ علیہ وسلم سے روایت نقل کرتے ہیں کہ آپ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جو بندہ کسی مصیبت میں مبتلا ہو چاہے وہ جانی ہو یا مالی اور وہ کسی آدمی سے اس مصیبت کا اظہار نہ کرے اور نہ ہی لوگوں سے شکوہ شکایت کرے تو اللہ تعالیٰ کے ذمہ ہے (بطور احسان و تفضل کے) کہ وہ اس کو بخش دیں۔
2|عَنْ أَبِيْ أُمَامَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: يَقُوْلُ اللّٰهُ سُبْحَانَهٗ: يَا ابْنَ آدَمَ إِنْ صَبَرْتَ وَاحْتَسَبْتَ عِنْدَ الصَّدْمَةِ الْأُوْلٰى لَمْ أَرْضَ لَكَ ثَوَابًا دُوْنَ الْجَنَّةِ.|سنن ابن ماجہ: رقم الحدیث 1597|يَا ابْنَ آدَمَ=اے آدم کے بیٹے~إِنْ=اگر~صَبَرْتَ=تو نے صبر کیا~احْتَسَبْتَ=تو نے ثواب کی امید رکھی~عِنْدَ=وقت~الصَّدْمَةِ=مصیبت~الْأُوْلٰى=پہلی~لَمْ أَرْضَ=میں پسند نہیں کروں گا~لَكَ=تیرے لیے~ثَوَابًا=اجر~دُوْنَ=کم~الْجَنَّةَ=جنت|حضرت ابو امامہ رضی اللہ تعالیٰ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ کا ارشاد ہے کہ اے ابن آدم! اگر تو نے صدمہ کے شروع میں میری رضا اور ثواب کی نیت سے صبر کیا تو میں تیرے لیے جنت سے کم ثواب پر راضی نہیں ہوں گا۔
3|عَنْ فَاطِمَةَ بِنْتِ الْحُسَيْنِ عَنْ أَبِيْهَا قَالَ: قَالَ النَّبِيُّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ أُصِيْبَ بِمُصِيْبَةٍ فَذَكَرَ مُصِيْبَتَهٗ فَأَحْدَثَ اسْتِرْجَاعًا وَإِنْ تَقَادَمَ عَهْدُهَا كَتَبَ اللّٰهُ لَهٗ مِنَ الْأَجْرِ مِثْلَهٗ يَوْمَ أُصِيْبَ.|سنن ابن ماجہ: حدیث نمبر 1600|أُصِيْبَ=اسے پہنچایا گیا~بِمُصِيْبَةٍ=کسی مصیبت کے ساتھ~فَذَكَرَ=پھر اس نے یاد کیا~مُصِيْبَتَهٗ=اس کی مصیبت~فَأَحْدَثَ=نئے سرے سے~اسْتِرْجَاعًا=انا للہ وانا الیہ راجعون پڑھنا~وَإِنْ=اگرچہ~تَقَادَمَ=پرانا ہو گیا~عَهْدُهَا=اس کا زمانہ~كَتَبَ=لکھا~لَهٗ=اس کے لیے~مِنَ الْأَجْرِ=بدلہ / ثواب~مِثْلَهٗ=اس کی مثل~يَوْمَ=دن~أُصِيْبَ=اسے مصیبت پہنچائی گئی|حضرت فاطمہ بنت حسین رحمہا اللہ اپنے والد حضرت حسین بن علی رضی اللہ عنہما سے روایت کرتی ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جس آدمی کو کوئی مصیبت پہنچی ہو (بعد میں) پھر اسے وہ مصیبت یاد آئی تو اس نے نئے سرے سے (إِنَّا لِلّٰهِ وَإِنَّا إِلَيْهِ رَاجِعُوْنَ) پڑھ لیا اگرچہ اس مصیبت کو گزرے طویل عرصہ ہو چکا ہو لیکن اللہ تعالیٰ اس کے لیے اتنا ہی ثواب لکھے گا جتنا اس دن ملا تھا جس دن مصیبت آئی تھی۔
# چھٹی فصل: شکر
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ لَا يَشْكُرِ النَّاسَ لَا يَشْكُرِ اللّٰهَ.|سنن الترمذی: رقم الحدیث 1954|مَنْ=جو شخص~لَا يَشْكُرِ=شکر ادا نہیں کرتا~النَّاسَ=لوگ~لَا يَشْكُرِ=شکر ادا نہیں کرے گا~اللّٰهَ=اللہ|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جو آدمی لوگوں کا شکر ادا نہیں کرتا وہ اللہ تعالیٰ کا شکر بھی ادا نہیں کرے گا۔
2|عَنْ صُهَيْبٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: عَجَبًا لِأَمْرِ الْمُؤْمِنِ إِنَّ أَمْرَهٗ كُلَّهٗ لَهٗ خَيْرٌ وَلَيْسَ ذٰلِكَ لِأَحَدٍ إِلَّا لِلْمُؤْمِنِ إِنْ أَصَابَتْهُ سَرَّاءُ شَكَرَ فَكَانَ خَيْرًا لَهٗ وَإِنْ أَصَابَتْهُ ضَرَّاءُ صَبَرَ فَكَانَ خَيْرًا لَهٗ.|صحیح مسلم: رقم الحدیث 7500|عَجَبًا=عجیب ہے~لِأَمْرِ=معاملہ~الْمُؤْمِنِ=ایمان والا~إِنَّ=بلاشبہ~أَمْرَهٗ=اس کا معاملہ~كُلَّهٗ=اس کا سب~لَهٗ=اس کے لیے~خَيْرٌ=بھلائی~لَيْسَ=نہیں ہے~ذٰلِكَ=وہ~لِأَحَدٍ=کسی کے لیے~إِلَّا=مگر~لِلْمُؤْمِنِ=ایمان والے کے لیے~سَرَّاءُ=خوشی~شَكَرَ=اس نے شکر کیا~فَكَانَ=پس وہ ہو گیا~خَيْرًا=بہتر~أَصَابَتْهُ=اس کو پہنچ گئی~ضَرَّاءُ=تکلیف~صَبَرَ=اس نے صبر کیا|حضرت صہیب رضی اللہ تعالیٰ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: مومن آدمی کا معاملہ بھی عجیب ہے، اس کے لیے ہر حال میں خیر ہی خیر ہے۔ اگر اس کو خوشی (راحت و آرام) پہنچے تو وہ اپنے رب کا شکر ادا کرتا ہے اور یہ اس کے لیے خیر ہے اور اگر اسے کوئی دکھ اور رنج پہنچتا ہے تو وہ اس پر صبر کرتا ہے اور یہ صبر کرنا بھی اس کے لیے سراسر خیر ہے۔
3|عَنْ عُمَرَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: يَا رَسُوْلَ اللّٰهِ! أَيَّ الْمَالِ نَتَّخِذُ؟ فَقَالَ: لِيَتَّخِذْ أَحَدُكُمْ قَلْبًا شَاكِرًا وَلِسَانًا ذَاكِرًا وَزَوْجَةً مُؤْمِنَةً تُعِيْنُ أَحَدَكُمْ عَلٰى أَمْرِ الْآخِرَةِ.|سنن ابن ماجہ: رقم الحدیث 1856|أَيَّ=کون سا~الْمَالِ=مال~نَتَّخِذُ=ہم بنائیں~لِيَتَّخِذْ=چاہیے کہ وہ بنائے~أَحَدُكُمْ=تم میں سے کوئی~قَلْبًا=دل~شَاكِرًا=شکر کرنے والا~لِسَانًا=زبان~ذَاكِرًا=ذکر کرنے والی~زَوْجَةً=بیوی~مُؤْمِنَةً=ایمان والی~تُعِيْنُ=وہ مدد کرتی ہے~عَلٰى=پر~أَمْرِ=معاملہ~الْآخِرَةِ=آخرت|حضرت عمر رضی اللہ عنہ سے روایت ہے کہ ایک دفعہ ہم نے عرض کیا: اے اللہ کے رسول! ہم کون سا مال رکھیں؟ آپ صلی اللہ علیہ وسلم نے فرمایا: تم میں سے ہر ایک کے پاس شکر کرنے والا دل، ذکر کرنے والی زبان اور ایمان والی بیوی ہونی چاہیے جو آخرت کے کاموں میں اس کی مدد کرے۔
# ساتویں فصل: امانت
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: آيَةُ الْمُنَافِقِ ثَلَاثٌ إِذَا حَدَّثَ كَذَبَ وَإِذَا وَعَدَ أَخْلَفَ وَإِذَا اؤْتُمِنَ خَانَ.|صحیح مسلم: رقم الحدیث 211|آيَةُ=نشانی~الْمُنَافِقِ=منافق~ثَلَاثٌ=تین~إِذَا=جب~حَدَّثَ=اس نے بات کی~كَذَبَ=اس نے جھوٹ بولا~وَعَدَ=اس نے وعدہ کیا~أَخْلَفَ=اس نے وعدہ خلافی کی~اؤْتُمِنَ=اس کے پاس امانت رکھوائی گئی~خَانَ=اس نے خیانت کی|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: منافق کی تین علامتیں ہیں جب بات کرے تو جھوٹ بولے، جب وعدہ کرے تو وعدہ کی خلاف ورزی کرے اور جب اس کے پاس امانت رکھوائی جائے تو اس میں خیانت کرے۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَلَّمَا خَطَبَنَا رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ إِلَّا قَالَ: لَا إِيْمَانَ لِمَنْ لَا أَمَانَةَ لَهٗ وَلَا دِيْنَ لِمَنْ لَا عَهْدَ لَهٗ.|مشکوٰۃ المصابیح: رقم الحدیث 35|لَا=کوئی نہیں~إِيْمَانَ=ایمان~لِمَنْ=اس شخص کے لیے~أَمَانَةَ=امانت داری~لَهٗ=اس کے لیے~دِيْنَ=دین~عَهْدَ=وعدہ|حضرت انس رضی اللہ عنہ فرماتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے ایسا خطبہ کم ہی دیا ہوگا جس میں یہ نہ فرمایا ہو کہ جس آدمی میں امانت نہیں اس کا ایمان کامل نہیں ہے اور جس میں ایفائے عہد نہیں اس کا دین بھی کامل نہیں ہے۔
3|عَنْ زَيْدِ بْنِ وَهْبٍ سَمِعْتُ حُذَيْفَةَ رَضِيَ اللّٰهُ عَنْهُ يَقُوْلُ: حَدَّثَنَا رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَنَّ الْأَمَانَةَ نَزَلَتْ مِنَ السَّمَاءِ فِيْ جَذْرِ قُلُوْبِ الرِّجَالِ وَنَزَلَ الْقُرْآنُ فَقَرَءُوا الْقُرْآنَ وَعَلِمُوْا مِنَ السُّنَّةِ.|صحیح البخاری: رقم الحدیث 7276|الْأَمَانَةَ=امانت~نَزَلَتْ=وہ اتری~مِنَ السَّمَاءِ=آسمان سے~فِيْ=میں~جَذْرِ=جڑ / رگ و ریشہ~قُلُوْبِ=دل~الرِّجَالِ=رجل کی جمع، لوگ~نَزَلَ=اترا~الْقُرْآنُ=قرآن~فَقَرَءُوا=پس انہوں نے پڑھا~عَلِمُوْا=انہوں نے جانا~مِنَ السُّنَّةِ=سنت / طریقہ کو|زید بن وہب رحمہ اللہ سے مروی ہے کہ میں نے حذیفہ بن یمان رضی اللہ عنہ سے سنا۔ وہ کہتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: امانت داری آسمان سے بعض لوگوں کی فطرت میں داخل ہوئی۔ جب قرآن مجید نازل ہوا تو انہوں نے قرآن مجید کو پڑھا اور سنت کو سیکھا۔
# آٹھویں فصل: صدق
1|عَنْ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِنَّ الصِّدْقَ يَهْدِيْ إِلَى الْبِرِّ وَإِنَّ الْبِرَّ يَهْدِيْ إِلَى الْجَنَّةِ وَإِنَّ الرَّجُلَ لَيَصْدُقُ حَتّٰى يَكُوْنَ صِدِّيْقًا وَإِنَّ الْكَذِبَ يَهْدِيْ إِلَى الْفُجُوْرِ وَإِنَّ الْفُجُوْرَ يَهْدِيْ إِلَى النَّارِ وَإِنَّ الرَّجُلَ لَيَكْذِبُ حَتّٰى يُكْتَبَ عِنْدَ اللّٰهِ كَذَّابًا.|صحیح البخاری: رقم الحدیث 6094|الصِّدْقَ=سچ بولنا~يَهْدِيْ=راہنمائی کرتا ہے~إِلٰى=طرف~الْبِرِّ=نیکی~الْجَنَّةِ=جنت~الرَّجُلَ=آدمی~لَيَصْدُقُ=البتہ سچ بولتا ہے~حَتّٰى=یہاں تک کہ~يَكُوْنَ=وہ ہو جاتا ہے~صِدِّيْقًا=مرتبہ صدیقیت کو پہنچنے والا~الْكَذِبَ=جھوٹ~الْفُجُوْرِ=گناہ~النَّارِ=آگ~لَيَكْذِبُ=البتہ جھوٹ بولتا ہے~يُكْتَبَ=اسے لکھا جاتا ہے~عِنْدَ اللّٰهِ=اللہ کے پاس~كَذَّابًا=بہت زیادہ جھوٹ بولنے والا|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے کہ نبی کریم صلی اللہ علیہ وسلم نے فرمایا: بے شک سچ نیکی کی طرف راہنمائی کرتا ہے اور نیکی جنت کی طرف لے جاتی ہے۔ ایک شخص سچ بولتا رہتا ہے یہاں تک کہ وہ صدیق کا لقب اور مرتبہ حاصل کر لیتا ہے اور بلاشبہ جھوٹ برائی کی طرف لے جاتا ہے اور برائی جہنم کی طرف، اور ایک شخص جھوٹ بولتا رہتا ہے یہاں تک کہ وہ اللہ کے یہاں بہت جھوٹا لکھ دیا جاتا ہے۔
2|عَنْ عُبَادَةَ بْنِ الصَّامِتِ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: اضْمَنُوْا لِيْ سِتًّا مِنْ أَنْفُسِكُمْ أَضْمَنْ لَكُمُ الْجَنَّةَ: أَدُّوْا إِذَا ائْتُمِنْتُمْ وَأَوْفُوْا إِذَا عَاهَدْتُّمْ وَاصْدُقُوْا إِذَا حَدَّثْتُمْ وَاحْفَظُوْا فُرُوْجَكُمْ وَغُضُّوْا أَبْصَارَكُمْ وَكُفُّوْا أَيْدِيَكُمْ.|شعب الایمان للبیہقی: رقم الحدیث 2556|اضْمَنُوْا=تم ضمانت دے دو~لِيْ=مجھے~سِتًّا=چھ~مِنْ=سے~أَنْفُسِكُمْ=تمہاری جانیں~أَضْمَنْ=میں ضمانت دیتا ہوں~لَكُمُ=تمہارے لیے~الْجَنَّةَ=جنت~أَدُّوْا=ادا کرو~ائْتُمِنْتُمْ=تمہارے پاس امانت رکھوائی جائے~أَوْفُوْا=تم پورا کرو~عَاهَدْتُّمْ=تم نے وعدہ کیا~اصْدُقُوْا=تم سچ بولو~حَدَّثْتُمْ=تم بات کرو~احْفَظُوْا=تم حفاظت کرو~فُرُوْجَكُمْ=اپنی شرمگاہوں کی~غُضُّوْا=جھکاؤ~أَبْصَارَكُمْ=اپنی آنکھوں کو~كُفُّوْا=تم روکو~أَيْدِيَكُمْ=اپنے ہاتھوں کو|حضرت عبادہ بن صامت رضی اللہ عنہ روایت کرتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: تم لوگ اپنے بارے میں مجھ سے چھ باتوں پر عمل کرنے کا عہد کر لو تو میں تمہیں جنت میں جانے کی ضمانت دیتا ہوں۔ جب تمہارے پاس امانت رکھی جائے تو امانت ادا کرو، وعدہ کرو تو پورا کرو، بات کرو تو سچ بولو، اپنی شرمگاہ کی حفاظت کرو، اپنی نگاہ کو محفوظ رکھو اور اپنے ہاتھوں پر قابو رکھو۔
3|عَنْ عَبْدِ الرَّحْمٰنِ بْنِ أَبِيْ قُرَادٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ تَوَضَّأَ يَوْمًا فَجَعَلَ أَصْحَابُهٗ يَتَمَسَّحُوْنَ بِوَضُوْئِهٖ فَقَالَ لَهُمُ النَّبِيُّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَا حَمَلَكُمْ عَلٰى هٰذَا؟ قَالُوْا: حُبُّ اللّٰهِ وَرَسُوْلِهٖ فَقَالَ النَّبِيُّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ سَرَّهٗ أَنْ يُحِبَّ اللّٰهَ وَرَسُوْلَهٗ أَوْ يُحِبَّهُ اللّٰهُ وَرَسُوْلُهٗ فَلْيَصْدُقْ حَدِيْثَهٗ إِذَا حَدَّثَ وَلْيُؤَدِّ أَمَانَتَهٗ إِذَا اؤْتُمِنَ وَلْيُحْسِنْ جِوَارَ مَنْ جَاوَرَهٗ.|شعب الایمان للبیہقی: رقم الحدیث 1533|تَوَضَّأَ=وضو کیا~يَوْمًا=ایک دن~فَجَعَلَ=شروع ہو گئے~أَصْحَابُهٗ=ان کے صحابہ~يَتَمَسَّحُوْنَ=جسم پر مل رہے تھے~بِوَضُوْئِهٖ=ان کے وضو کا پانی~مَا حَمَلَكُمْ=کس چیز نے آپ کو ابھارا ہے~عَلٰى هٰذَا=اس پر~قَالُوْا=وہ کہنے لگے~حُبُّ=محبت~رَسُوْلِهٖ=اس کا رسول~سَرَّهٗ=اس کو خوش کرے~أَنْ يُحِبَّ=کہ وہ محبت کرے~أَوْ=یا~يُحِبَّهٗ=وہ اس سے محبت کرے~فَلْيَصْدُقْ=اس کو چاہیے کہ وہ سچ بولے~حَدِيْثَهٗ=اس کی بات~حَدَّثَ=وہ بات کرے~لْيُؤَدِّ=چاہیے کہ وہ ادا کر دے~أَمَانَتَهٗ=اس کی امانت~اؤْتُمِنَ=اس کے پاس امانت رکھوائی جائے~لْيُحْسِنْ=چاہیے کہ اچھائی کرے~جِوَارَ=پڑوس~مَنْ جَاوَرَهٗ=جو شخص اس کا پڑوسی بنا|حضرت عبد الرحمن بن ابی قراد رضی اللہ عنہ سے روایت ہے کہ ایک دن نبی اکرم صلی اللہ علیہ وسلم نے وضو کیا تو صحابہ کرام رضوان اللہ علیہم اجمعین نے حضور صلی اللہ علیہ وسلم کے وضو کے پانی کو اپنے جسم پر ملنا شروع کر دیا۔ نبی اکرم صلی اللہ علیہ وسلم نے یہ دیکھ کر فرمایا: تم یہ جو کچھ کر رہے ہو اس کا سبب کیا ہے؟ انہوں نے عرض کیا: اللہ اور اس کے رسول کی محبت اس کا باعث ہے۔ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: جو شخص یہ پسند کرتا ہے کہ وہ اللہ اور اس کے رسول سے محبت کرے یا اللہ اور اس کا رسول اس سے محبت کریں تو اس کو چاہیے کہ جب بولے تو سچ بولے، جب اس کے پاس امانت رکھی جائے تو اس امانت کو ادا کرے اور پڑوسیوں کے ساتھ اچھی ہمسائیگی کا ثبوت دے۔
# نویں فصل: زہد
1|عَنْ جَابِرِ بْنِ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَيُّهَا النَّاسُ اتَّقُوا اللّٰهَ وَأَجْمِلُوْا فِي الطَّلَبِ فَإِنَّ نَفْسًا لَنْ تَمُوْتَ حَتّٰى تَسْتَوْفِيَ رِزْقَهَا وَإِنْ أَبْطَأَ عَنْهَا فَاتَّقُوا اللّٰهَ وَأَجْمِلُوْا فِي الطَّلَبِ خُذُوْا مَا حَلَّ وَدَعُوْا مَا حَرُمَ.|سنن ابن ماجہ: رقم الحدیث 2144|أَيُّهَا النَّاسُ=اے لوگو~اتَّقُوا اللّٰهَ=تم اللہ سے ڈرو~أَجْمِلُوْا=تم اچھائی کرو (یعنی اعتدال سے کام کرو)~فِي الطَّلَبِ=تلاش کرنا / مانگنا~فَإِنَّ=پس بلاشبہ~نَفْسًا=ذات~لَنْ تَمُوْتَ=ہرگز نہیں مرے گا~حَتّٰى=یہاں تک کہ~تَسْتَوْفِيَ=وہ پورا وصول کرے~رِزْقَهَا=اس کا رزق~وَإِنْ=اگرچہ~أَبْطَأَ=اس نے دیر کی~عَنْهَا=اس کے بارے میں~خُذُوْا=تم پکڑ لو~مَا=جو چیز~حَلَّ=وہ حلال ہوا~دَعُوْا=اور تم چھوڑ دو~حَرُمَ=وہ حرام ہو گیا|حضرت جابر بن عبد اللہ رضی اللہ عنہ کہتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اے لوگو! اللہ سے ڈرو اور دنیا کے طلب کرنے میں اعتدال کا راستہ اختیار کرو۔ کوئی آدمی اس وقت تک فوت نہیں ہوگا جب تک اپنا رزق پورا نہ کر لے۔ گو کہ ممکن ہے کہ اس میں تاخیر ہو جائے۔ لہٰذا اللہ تعالیٰ سے ڈرو اور روزی کے طلب کرنے میں اعتدال کا راستہ اختیار کرو۔ جو رزق حلال ہو اسے لے لو اور جو حرام ہو اسے چھوڑ دو۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ كَانَتْ نِيَّتُهٗ طَلَبَ الْآخِرَةِ جَعَلَ اللّٰهُ غِنَاهُ فِيْ قَلْبِهٖ وَجَمَعَ لَهٗ شَمْلَهٗ وَأَتَتْهُ الدُّنْيَا وَهِيَ رَاغِمَةٌ وَمَنْ كَانَتْ نِيَّتُهٗ طَلَبَ الدُّنْيَا جَعَلَ اللّٰهُ الْفَقْرَ بَيْنَ عَيْنَيْهِ وَشَتَّتَ عَلَيْهِ أَمْرَهٗ وَلَا يَأْتِيْهِ مِنْهَا إِلَّا مَا كُتِبَ لَهٗ.|سنن الترمذی: رقم الحدیث 2465|كَانَتْ=وہ تھی~نِيَّتُهٗ=اس کی نیت~طَلَبَ=طلب کرنا~الْآخِرَةِ=آخرت~جَعَلَ=اس نے کیا~غِنَاهُ=اس کی بے نیازی~فِيْ قَلْبِهٖ=اس کا دل~جَمَعَ=اس نے جمع کیا~شَمْلَهٗ=اس کا شیرازہ~أَتَتْهُ=اس کے پاس آئی~الدُّنْيَا=دنیا~هِيَ=وہ~رَاغِمَةٌ=ناک زمین پر رگڑنے والی / ذلیل ہونے والی~الْفَقْرَ=تنگدستی~بَيْنَ=درمیان~عَيْنَيْهِ=اس کی دو آنکھیں~شَتَّتَ=اس نے بکھیر دیا~عَلَيْهِ=اس پر~أَمْرَهٗ=اس کا معاملہ~لَا يَأْتِيْهِ=اور اس کے پاس نہیں آئے گا~مِنْهَا=اس میں سے~إِلَّا=مگر~مَا=جو~كُتِبَ=لکھ دیا گیا~لَهٗ=اس کے لیے|حضرت انس بن مالک رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جس شخص کی نیت آخرت کی طلب کی ہو تو اللہ تعالیٰ اس کے دل کو غنی نصیب فرمائیں گے اور اس کے پراگندہ حال کو درست فرما دیں گے اور دنیا اس کے پاس خود بخود ذلیل ہو کر آئے گی۔ جس شخص کی نیت دنیا کو طلب کرنے کی ہوگی؛ اللہ تعالیٰ محتاجی کے آثار اس کی دونوں آنکھوں کے درمیان اس کے چہرے پر پیدا کر دیں گے اور اس کے حال کو پراگندہ کر دیں گے اور یہ دنیا اس کو بس اسی قدر ملے گی جس قدر اس کے واسطے پہلے سے لکھی جا چکی ہوگی۔
3|عَنْ مَحْمُوْدِ بْنِ لَبِيْدٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِنَّ اللّٰهَ يَحْمِيْ عَبْدَهُ الْمُؤْمِنَ فِي الدُّنْيَا وَهُوَ يُحِبُّهٗ كَمَا تَحْمُوْنَ مَرِيْضَكُمُ الطَّعَامَ وَالشَّرَابَ تَخَافُوْنَ عَلَيْهِ.|مسند احمد: رقم الحدیث 23518|يَحْمِيْ=حفاظت کرتا ہے~عَبْدَهُ=اس کا بندہ~الْمُؤْمِنَ=مومن~الدُّنْيَا=دنیا~هُوَ=وہ~يُحِبُّهٗ=اس سے محبت کرتا ہے~كَمَا=جس طرح~تَحْمُوْنَ=تم نگہداشت کرتے ہو~مَرِيْضَكُمُ=تمہارے بیمار~الطَّعَامَ=کھانا~الشَّرَابَ=پینا~تَخَافُوْنَ=تم خوف کھاتے ہو~عَلَيْهِ=اس پر|حضرت محمود بن لبید رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اللہ تعالیٰ اپنے (محبوب) مومن بندے کو دنیا (کی رنگینیوں اور آرائشوں سے) اس طرح بچاتا ہے جیسے تم لوگ اپنے مریض کو کھانے پینے کی اشیاء سے اس اندیشے سے بچاتے ہو کہ اس کی صحت خراب نہ ہو جائے۔
# دسویں فصل: توکل
1|عَنْ عُمَرَ بْنِ الْخَطَّابِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: لَوْ أَنَّكُمْ كُنْتُمْ تَوَكَّلُوْنَ عَلَى اللّٰهِ حَقَّ تَوَكُّلِهٖ لَرُزِقْتُمْ كَمَا تُرْزَقُ الطَّيْرُ تَغْدُوْ خِمَاصًا وَتَرُوْحُ بِطَانًا.|سنن الترمذی: رقم الحدیث 2344|لَوْ=اگر~أَنَّكُمْ=بے شک تم~كُنْتُمْ تَوَكَّلُوْنَ=تم توکل کرو~عَلَى اللّٰهِ=اللہ پر~حَقَّ=حق ہے~تَوَكُّلِهٖ=اس پر توکل کرنا~لَرُزِقْتُمْ=تمہیں ضرور رزق دیا جائے گا~كَمَا=جیسا کہ~تُرْزَقُ=رزق دیا جاتا ہے~الطَّيْرُ=پرندے~تَغْدُوْ=وہ صبح نکلتے ہیں~خِمَاصًا=خالی پیٹ~تَرُوْحُ=شام کو آتے ہیں~بِطَانًا=پیٹ بھرا ہونا|حضرت عمر بن خطاب رضی اللہ عنہ کہتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اگر تم لوگ اللہ تعالیٰ پر توکل کرو جیسا کہ اس پر توکل کرنے کا حق ہے تو تمہیں اسی طرح رزق ملے گا جس طرح کہ پرندوں کو ملتا ہے کہ وہ صبح کو بھوکے نکلتے ہیں اور شام کو پیٹ بھر کر واپس آتے ہیں۔
2|عَنْ عَبْدِ اللّٰهِ بْنِ مَسْعُوْدٍ رَضِيَ اللّٰهُ عَنْهُ عَنْ رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: اَلطِّيَرَةُ شِرْكٌ اَلطِّيَرَةُ شِرْكٌ ثَلَاثًا وَمَا مِنَّا إِلَّا وَلٰكِنَّ اللّٰهَ يُذْهِبُهٗ بِالتَّوَكُّلِ.|سنن ابی داؤد: رقم الحدیث 3910|الطِّيَرَةُ=بدشگونی~شِرْكٌ=شرک کرنا~ثَلَاثًا=تین مرتبہ~مَا=نہیں~مِنَّا=ہم میں سے~إِلَّا=مگر~لٰكِنَّ اللّٰهَ=لیکن اللہ~يُذْهِبُهٗ=وہ اسے ختم کر دے گا~بِالتَّوَكُّلِ=توکل کی وجہ سے|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے تین بار ارشاد فرمایا: بدشگونی شرک ہے اور ہم میں سے ہر ایک کو جو وہم ہو جاتا ہے اللہ تعالیٰ اس کو توکل سے دور فرما دیتے ہیں۔ (یعنی اللہ تعالیٰ کی ذات پر توکل کرنے کی وجہ سے یہ وہم ختم ہو جاتا ہے۔)
3|عَنْ أَنَسِ بْنِ مَالِكٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِذَا خَرَجَ الرَّجُلُ مِنْ بَيْتِهٖ فَقَالَ: بِسْمِ اللّٰهِ تَوَكَّلْتُ عَلَى اللّٰهِ لَا حَوْلَ وَلَا قُوَّةَ إِلَّا بِاللّٰهِ قَالَ: يُقَالُ حِيْنَئِذٍ: هُدِيْتَ وَكُفِيْتَ وَوُقِيْتَ فَتَتَنَحّٰى لَهُ الشَّيَاطِيْنُ فَيَقُوْلُ لَهٗ شَيْطَانٌ آخَرُ: كَيْفَ لَكَ بِرَجُلٍ قَدْ هُدِيَ وَكُفِيَ وَوُقِيَ؟|سنن ابی داؤد: رقم الحدیث 5095|إِذَا=جب~خَرَجَ=نکلا~الرَّجُلُ=آدمی~مِنْ بَيْتِهٖ=اس کا گھر~فَقَالَ=پس اس نے کہا~بِسْمِ اللّٰهِ=اللہ کے نام کے ساتھ~تَوَكَّلْتُ=میں نے بھروسہ کیا~عَلَى اللّٰهِ=اللہ پر~لَا حَوْلَ=نہیں نیکی کرنے کی طاقت~لَا قُوَّةَ=نہیں گناہ سے بچنے کی قوت~إِلَّا بِاللّٰهِ=مگر اللہ کی مدد سے~يُقَالُ=کہا جاتا ہے~حِيْنَئِذٍ=اس وقت~هُدِيْتَ=تجھے ہدایت دی گئی~كُفِيْتَ=تیری کفایت کی گئی~وُقِيْتَ=تیری حفاظت کی گئی~فَتَتَنَحّٰى=پس الگ ہو جاتا ہے~لَهُ=اس سے~الشَّيَاطِيْنُ=شیطان کی جمع~شَيْطَانٌ آخَرُ=دوسرا شیطان~كَيْفَ لَكَ=کیسا ہوگا تیرے لیے~بِرَجُلٍ=ایک آدمی کے ساتھ~هُدِيَ=اس کی رہنمائی کی گئی~كُفِيَ=اس کی کفایت کی گئی~وُقِيَ=اس کی حفاظت کی گئی|حضرت انس بن مالک رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جب آدمی اپنے گھر سے یہ کہتے ہوئے نکلے: بِسْمِ اللّٰهِ تَوَكَّلْتُ عَلَى اللّٰهِ لَا حَوْلَ وَلَا قُوَّةَ إِلَّا بِاللّٰهِ (اللہ تعالیٰ پر توکل کرتے ہوئے اس کے نام کی برکت کے ساتھ میں گھر سے نکلتا ہوں، نیکی کرنے کی طاقت اور گناہ سے بچنے کی ہمت اللہ تعالیٰ کی توفیق سے ہوتی ہے) آپ صلی اللہ علیہ وسلم نے فرمایا کہ اس وقت کہا جاتا ہے: اب تجھے ہدایت دے دی گئی، تیری کفایت کر دی گئی اور تو بچا لیا گیا، (یہ سن کر) شیطان اس سے جدا ہو جاتا ہے، تو اس سے دوسرا شیطان کہتا ہے: تیرا داؤ بھلا ایسے آدمی پر کیسے چلے گا جسے ہدایت دی گئی ہے، جس کی کفایت کر دی گئی ہے اور جسے بچا لیا گیا ہے۔
## اَلْبَابُ الرَّابِعُ: اَلْمُعَاشَرَاتُ — معاشرات
# پہلی فصل: زوجین کے حقوق
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَوْ كُنْتُ آمِرًا أَحَدًا أَنْ يَسْجُدَ لِأَحَدٍ لَأَمَرْتُ الْمَرْأَةَ أَنْ تَسْجُدَ لِزَوْجِهَا.|سنن الترمذی: رقم الحدیث 1159|لَوْ=اگر~كُنْتُ=میں ہوتا~آمِرًا=حکم دینے والا~أَحَدًا=کسی ایک کو~أَنْ يَسْجُدَ=یہ کہ وہ سجدہ کرے~لِأَحَدٍ=کسی ایک کو~لَأَمَرْتُ=تو میں حکم دیتا~الْمَرْأَةَ=عورت~أَنْ تَسْجُدَ=یہ کہ وہ سجدہ کرے~لِزَوْجِهَا=اپنے شوہر کو|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: اگر میں مخلوق میں سے کسی کو (کسی اور کو) سجدے کا حکم دیتا تو عورت کو حکم دیتا کہ وہ اپنے خاوند کو سجدہ کرے۔
2|عَنْ أُمِّ سَلَمَةَ رَضِيَ اللّٰهُ عَنْهَا قَالَتْ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَيُّمَا امْرَأَةٍ مَاتَتْ وَزَوْجُهَا عَنْهَا رَاضٍ دَخَلَتِ الْجَنَّةَ.|سنن الترمذی: رقم الحدیث 1161|أَيُّمَا=جو کوئی~امْرَأَةٍ=عورت~مَاتَتْ=فوت ہو گئی~زَوْجُهَا=اس کا شوہر~رَاضٍ=راضی~دَخَلَتْ=وہ داخل ہوگی|حضرت ام سلمہ رضی اللہ عنہا سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جو عورت اس حالت میں فوت ہوئی کہ اس کا شوہر اس سے راضی تھا تو وہ جنت میں جائے گی۔
3|عَنْ عَائِشَةَ رَضِيَ اللّٰهُ عَنْهَا قَالَتْ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ مِنْ أَكْمَلِ الْمُؤْمِنِيْنَ إِيْمَانًا أَحْسَنُهُمْ خُلُقًا وَأَلْطَفُهُمْ بِأَهْلِهٖ.|مسند احمد: رقم الحدیث 24086|أَكْمَلِ=سب سے مکمل~أَحْسَنُهُمْ=ان میں سب سے بہتر~أَلْطَفُهُمْ=ان میں سب سے مہربان~بِأَهْلِهٖ=اپنی اہلیہ کے ساتھ|حضرت عائشہ صدیقہ رضی اللہ عنہا سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: مسلمانوں میں اس آدمی کا ایمان زیادہ کامل ہے جس کا اخلاق بہت اچھا ہو اور جس کا رویہ بیوی کے ساتھ لطف و محبت کا ہو۔
# دوسری فصل: والدین کے حقوق
1|عَنْ عَبْدِ اللّٰهِ بْنِ عَمْرٍو رَضِيَ اللّٰهُ عَنْهُمَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: رِضَى الرَّبِّ فِيْ رِضَى الْوَالِدِ وَسَخَطُ الرَّبِّ فِيْ سَخَطِ الْوَالِدِ.|سنن الترمذی: رقم الحدیث 1899|رِضَى=رضامندی~الرَّبِّ=رب (اللہ تعالیٰ)~الْوَالِدِ=باپ~سَخَطُ=ناراضی|حضرت عبد اللہ بن عمرو (بن العاص) رضی اللہ عنہما سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: اللہ کی رضامندی والد کی رضامندی میں ہے اور اللہ کی ناراضگی والد کی ناراضگی میں ہے۔
2|عَنْ أَبِيْ أُمَامَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَجُلًا قَالَ: يَا رَسُوْلَ اللّٰهِ! مَا حَقُّ الْوَالِدَيْنِ عَلٰى وَلَدِهِمَا؟ قَالَ: هُمَا جَنَّتُكَ وَنَارُكَ.|سنن ابن ماجہ: رقم الحدیث 3662|مَا حَقُّ=کیا حق ہے~الْوَالِدَيْنِ=والدین~عَلٰى وَلَدِهِمَا=ان کی اولاد پر~هُمَا=وہ دونوں~جَنَّتُكَ=تیری جنت~نَارُكَ=تیری آگ|حضرت ابو امامہ رضی اللہ عنہ سے روایت ہے کہ ایک شخص نے (رسول اللہ صلی اللہ علیہ وسلم سے) پوچھا: اے اللہ کے رسول! اولاد پر ماں باپ کا کتنا حق ہے؟ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: وہ (والدین) تمہاری جنت اور دوزخ ہیں۔
3|عَنِ ابْنِ عُمَرَ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: بِرُّوْا آبَاءَكُمْ يَبَرَّكُمْ أَبْنَاؤُكُمْ وَعِفُّوْا تَعِفُّ نِسَاؤُكُمْ.|کنز العمال فی سنن الاقوال والافعال: رقم الحدیث 45468|بِرُّوْا=حسن سلوک کرو~آبَاءَكُمْ=اپنے باپ دادا~يَبَرَّكُمْ=تمہاری فرمانبرداری کریں گے~أَبْنَاؤُكُمْ=تمہارے بیٹے~عِفُّوْا=تم پاکدامن رہو~تَعِفُّ=پاک دامن رہیں گی~نِسَاؤُكُمْ=تمہاری عورتیں|حضرت عبد اللہ بن عمر رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: اپنے ماں باپ کی خدمت و فرمانبرداری کرو، تمہاری اولاد تمہاری فرمانبردار اور خدمت گزار ہوگی اور تم پاک دامنی کے ساتھ رہو تمہاری عورتیں پاک دامن رہیں گی۔
# تیسری فصل: اولاد کے حقوق
1|عَنْ عُبَيْدِ اللّٰهِ بْنِ أَبِيْ رَافِعٍ عَنْ أَبِيْهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: رَأَيْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ أَذَّنَ فِيْ أُذُنِ الْحَسَنِ بْنِ عَلِيٍّ حِيْنَ وَلَدَتْهُ فَاطِمَةُ بِالصَّلَاةِ.|سنن ابی داؤد: رقم الحدیث 5105|رَأَيْتُ=میں نے دیکھا~أَذَّنَ=اس نے اذان دی~فِيْ أُذُنِ=کان میں~الْحَسَنِ بْنِ عَلِيٍّ=حضرت حسن بن علی~حِيْنَ=جب~وَلَدَتْهُ=اس کو جنا~فَاطِمَةُ=حضرت فاطمہ~بِالصَّلَاةِ=نماز والی|حضرت عبید اللہ بن ابی رافع اپنے والد حضرت ابو رافع رضی اللہ عنہ سے روایت کرتے ہیں کہ میں نے رسول اللہ صلی اللہ علیہ وسلم کو حضرت حسن بن علی رضی اللہ عنہما کے کان میں نماز کے لیے دی جانے والی اذان جیسی اذان دیتے ہوئے دیکھا جب حضرت فاطمہ رضی اللہ عنہا کے ہاں ان کی ولادت ہوئی تھی۔
2|عَنِ الْحَسَنِ عَنْ سَمُرَةَ بْنِ جُنْدُبٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: كُلُّ غُلَامٍ رَهِيْنَةٌ بِعَقِيْقَتِهٖ تُذْبَحُ عَنْهُ يَوْمَ سَابِعِهٖ وَيُحْلَقُ وَيُسَمّٰى.|سنن ابی داؤد: رقم الحدیث 2838|كُلُّ=ہر ایک~غُلَامٍ=بچہ~رَهِيْنَةٌ=گروی رکھا ہوا~بِعَقِيْقَتِهٖ=اس کے عقیقہ کے بدلے~تُذْبَحُ=ذبح کیا جائے گا~عَنْهُ=اس کی طرف سے~يَوْمَ سَابِعِهٖ=اس کے ساتویں دن~يُحْلَقُ=اس کا سر منڈوایا جائے~يُسَمّٰى=اس کا نام رکھا جائے|حضرت حسن رحمہ اللہ نے حضرت سمرہ بن جندب رضی اللہ عنہ سے روایت کیا ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: ہر بچہ اپنے عقیقہ کے جانور کے بدلہ میں گروی رکھا ہوتا ہے جو ساتویں دن اس کی طرف سے ذبح کیا جائے اور (اسی دن) اس کا سر منڈوا دیا جائے اور نام رکھا جائے۔
3|عَنْ عَائِشَةَ رَضِيَ اللّٰهُ عَنْهَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: حَقُّ الْوَلَدِ عَلٰى وَالِدِهٖ أَنْ يُحْسِنَ اسْمَهٗ وَيُحْسِنَ مِنْ مُرْضِعِهٖ وَيُحْسِنَ أَدَبَهٗ.|شعب الایمان للبیہقی: رقم الحدیث 8667|حَقُّ=حق~الْوَلَدِ=بیٹا~عَلٰى وَالِدِهٖ=اس کے باپ پر~أَنْ يُحْسِنَ=کہ وہ اچھا رکھے~اسْمَهٗ=اس کا نام~مُرْضِعِهٖ=اس کو دودھ پلانے والی~أَدَبَهٗ=اس کا ادب|حضرت عائشہ رضی اللہ عنہا سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: باپ پر بچے کا یہ حق ہے کہ اس کا اچھا نام رکھے اور اس کے دودھ پلانے کا بہترین انتظام کرے اور اس کو حسن ادب سکھائے۔
# چوتھی فصل: رشتہ داروں کے حقوق
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: تَعَلَّمُوْا مِنْ أَنْسَابِكُمْ مَا تَصِلُوْنَ بِهٖ أَرْحَامَكُمْ فَإِنَّ صِلَةَ الرَّحِمِ مَحَبَّةٌ فِي الْأَهْلِ مَثْرَاةٌ فِي الْمَالِ مَنْسَأَةٌ فِي الْأَثَرِ.|سنن الترمذی: رقم الحدیث 1979|تَعَلَّمُوْا=تم سیکھو~مِنْ=سے~أَنْسَابِكُمْ=تمہارے نسب~مَا=جو~تَصِلُوْنَ=تم جوڑتے ہو~بِهٖ=اس کے ساتھ~أَرْحَامَكُمْ=تمہاری رشتہ داریاں~فَإِنَّ=پس بے شک~صِلَةَ الرَّحِمِ=رشتہ داری جوڑنا~مَحَبَّةٌ=محبت~فِي الْأَهْلِ=اہل میں~مَثْرَاةٌ=کثرت کا ذریعہ~فِي الْمَالِ=مال میں~مَنْسَأَةٌ=دیر کا سبب~فِي الْأَثَرِ=زندگی / عمر|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: اپنے نسب کو پہچانو تاکہ اس کے ذریعے تم اپنے رشتے داروں سے صلہ رحمی کر سکو۔ کیونکہ صلہ رحمی رشتہ داروں سے محبت بڑھنے کا سبب ہے، مال و دولت میں اضافے کا سبب ہے اور آدمی کی عمر میں اضافے کا سبب ہے۔
2|عَنْ أَنَسٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ فِيْ مَرَضِهٖ: أَرْحَامَكُمْ أَرْحَامَكُمْ.|صحیح ابن حبان: رقم الحدیث 4361|فِيْ مَرَضِهٖ=اس کا مرض~أَرْحَامَكُمْ=تمہاری رشتہ داریاں|حضرت انس بن مالک رضی اللہ عنہ سے مروی ہے کہ رسول اکرم صلی اللہ علیہ وسلم نے اپنی بیماری کے دوران ارشاد فرمایا: اپنے رشتے داروں کا خیال رکھو، اپنے رشتے داروں کا خیال رکھو۔
3|عَنْ عَائِشَةَ رَضِيَ اللّٰهُ عَنْهَا أَنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِنَّهٗ مَنْ أُعْطِيَ حَظَّهٗ مِنَ الرِّفْقِ فَقَدْ أُعْطِيَ حَظَّهٗ مِنْ خَيْرِ الدُّنْيَا وَالْآخِرَةِ وَصِلَةُ الرَّحِمِ وَحُسْنُ الْخُلُقِ وَحُسْنُ الْجِوَارِ يَعْمُرَانِ الدِّيَارَ وَيَزِيْدَانِ فِي الْأَعْمَارِ.|مسند احمد: رقم الحدیث 25135|أُعْطِيَ=اسے دیا گیا~حَظَّهٗ=اس کا حصہ~مِنَ الرِّفْقِ=نرمی~صِلَةُ=ملانا~الرَّحِمِ=رشتہ داری~حُسْنُ=اچھا~الْخُلُقِ=اخلاق~الْجِوَارِ=پڑوسی~يَعْمُرَانِ=وہ آباد کرتے ہیں~الدِّيَارَ=گھر~يَزِيْدَانِ=وہ دونوں اضافہ کرتے ہیں~فِي الْأَعْمَارِ=عمروں میں|حضرت عائشہ رضی اللہ عنہا سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: جسے نرمی میں سے ایک حصہ دے دیا گیا اسے دنیا و آخرت کی بھلائی میں سے ایک حصہ دے دیا گیا اور صلہ رحمی، حسن خلق اور اچھا پڑوسی بننا گھروں کو آباد کرتا ہے اور عمروں میں اضافہ کرتا ہے۔
# پانچویں فصل: پڑوسیوں کے حقوق
1|عَنْ أَنَسِ بْنِ مَالِكٍ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَيْسَ بِمُؤْمِنٍ مَنْ لَا يَأْمَنُ جَارُهٗ غَوَائِلَهٗ.|مستدرک حاکم: رقم الحدیث 7380|لَيْسَ=نہیں ہے~بِمُؤْمِنٍ=مومن~مَنْ=جو شخص~لَا يَأْمَنُ=محفوظ نہ ہو~جَارُهٗ=اس کا پڑوسی~غَوَائِلَهٗ=اس کی شرارتیں|حضرت انس بن مالک رضی اللہ عنہ سے روایت ہے رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: وہ شخص کامل مومن نہیں ہو سکتا جس کا پڑوسی اس کی شرارتوں سے محفوظ نہ ہو۔
2|عَنِ ابْنِ عَبَّاسٍ رَضِيَ اللّٰهُ عَنْهُمَا قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ بَنٰى بِنَاءً فَلْيَدْعَمْهُ حَائِطَ جَارِهٖ.|مسند احمد: رقم الحدیث 2098|مَنْ=جو شخص~بَنٰى=اس نے بنایا~بِنَاءً=عمارت~فَلْيَدْعَمْهُ=پس چاہیے کہ اسے سہارا دے~حَائِطَ=دیوار~جَارِهٖ=اس کا پڑوسی|حضرت عبد اللہ بن عباس رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جو شخص کوئی عمارت بنائے تو اسے چاہیے کہ وہ اپنے پڑوسی کی دیوار کے لیے (اگر وہ اس پر کمرہ بنانا چاہے تو) اسے سہارا دے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ كَانَ يُؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْآخِرِ فَلَا يُؤْذِيْ جَارَهٗ.|صحیح البخاری: رقم الحدیث 5185|يُؤْمِنُ=ایمان رکھتا ہے~بِاللّٰهِ=اللہ پر~الْيَوْمِ الْآخِرِ=قیامت کا دن~فَلَا يُؤْذِيْ=پس چاہیے کہ وہ تکلیف نہ دے~جَارَهٗ=اس کا پڑوسی|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ نبی کریم صلی اللہ علیہ وسلم نے فرمایا: جو شخص اللہ تعالیٰ اور قیامت کے دن پر ایمان رکھتا ہو وہ اپنے پڑوسی کو تکلیف نہ پہنچائے۔
# چھٹی فصل: یتیموں کے حقوق
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ أَنَّ رَجُلًا شَكَا إِلٰى رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَسْوَةَ قَلْبِهٖ فَقَالَ لَهٗ: إِنْ أَرَدْتَّ تَلْيِيْنَ قَلْبِكَ فَأَطْعِمِ الْمِسْكِيْنَ وَامْسَحْ رَأْسَ الْيَتِيْمِ.|مسند احمد: رقم الحدیث 7566|شَكَا=اس نے شکایت کی~إِلٰى=طرف~قَسْوَةَ=سختی~قَلْبِهٖ=اس کا دل~أَرَدْتَّ=تو چاہے~تَلْيِيْنَ=نرم کرنا~قَلْبِكَ=تیرا دل~فَأَطْعِمِ=پس کھلا~الْمِسْكِيْنَ=مسکین~امْسَحْ=ہاتھ پھیر~رَأْسَ=سر~الْيَتِيْمِ=یتیم|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ ایک آدمی نے رسول اللہ صلی اللہ علیہ وسلم سے شکایت کی کہ اس کا دل بہت سخت ہے۔ آپ صلی اللہ علیہ وسلم نے اس سے فرمایا: اگر تم چاہتے ہو کہ تمہارا دل نرم ہو جائے تو مسکین کو کھانا کھلاؤ اور یتیم کے سر پر ہاتھ پھیرو۔
2|عَنْ سَهْلٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَنَا وَكَافِلُ الْيَتِيْمِ فِي الْجَنَّةِ هٰكَذَا وَأَشَارَ بِالسَّبَّابَةِ وَالْوُسْطٰى وَفَرَّجَ بَيْنَهُمَا شَيْئًا.|صحیح البخاری: رقم الحدیث 5304|كَافِلُ=کفالت کرنے والا~الْيَتِيْمِ=یتیم~فِي الْجَنَّةِ=جنت میں~هٰكَذَا=اس طرح~أَشَارَ=اشارہ کیا~بِالسَّبَّابَةِ=انگشت شہادت سے~الْوُسْطٰى=درمیانی انگلی~فَرَّجَ=فاصلہ رکھا~بَيْنَهُمَا=ان دونوں کے درمیان~شَيْئًا=تھوڑا سا|حضرت سہل رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: میں اور یتیم کی کفالت کرنے والا جنت میں ان دو انگلیوں کی طرح ہوں گے۔ آپ نے اپنی شہادت اور درمیان والی انگلی سے اشارہ کیا اور دونوں میں تھوڑا سا فاصلہ رکھا۔
3|عَنْ عَدِيِّ بْنِ حَاتِمٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ ضَمَّ يَتِيْمًا لَهٗ أَوْ لِغَيْرِهٖ حَتّٰى يُغْنِيَهُ اللّٰهُ عَنْهُ وَجَبَتْ لَهُ الْجَنَّةُ.|المعجم الاوسط للطبرانی: رقم الحدیث 5345|ضَمَّ=ملایا~يَتِيْمًا=یتیم~لِغَيْرِهٖ=اس کے غیر کا~حَتّٰى=یہاں تک کہ~يُغْنِيَهُ=اس کو بے نیاز کر دے~وَجَبَتْ=واجب ہو گئی|حضرت عدی بن حاتم رضی اللہ عنہ سے روایت ہے رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جس شخص نے اپنے یتیم یا غیر کے یتیم کی دیکھ بھال کی یہاں تک کہ اللہ تعالیٰ نے اسے اس سے مستغنی کر دیا (وہ یتیم خود اپنے قدموں پر کھڑا ہو گیا) تو اس (یتیم کی پرورش کرنے والے) کے لیے جنت واجب ہو گئی۔
# ساتویں فصل: مہمان کے حقوق
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: مَنْ كَانَ يُؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْآخِرِ فَلَا يُؤْذِ جَارَهٗ وَمَنْ كَانَ يُؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْآخِرِ فَلْيُكْرِمْ ضَيْفَهٗ وَمَنْ كَانَ يُؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْآخِرِ فَلْيَقُلْ خَيْرًا أَوْ لِيَصْمُتْ.|صحیح البخاری: رقم الحدیث 6018|يُؤْمِنُ=ایمان لاتا ہے~بِاللّٰهِ=اللہ پر~فَلَا يُؤْذِ=پس وہ تکلیف نہ دے~جَارَهٗ=اس کا پڑوسی~فَلْيُكْرِمْ=پس چاہیے کہ وہ اکرام کرے~ضَيْفَهٗ=اس کا مہمان~فَلْيَقُلْ=پس چاہیے کہ وہ کہے~خَيْرًا=بھلائی کی بات~أَوْ=یا~لِيَصْمُتْ=وہ خاموش رہے|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: جو کوئی اللہ اور آخرت کے دن پر ایمان رکھتا ہو وہ اپنے پڑوسی کو تکلیف نہ پہنچائے اور جو کوئی اللہ اور آخرت کے دن پر ایمان رکھتا ہو وہ اپنے مہمان کا اکرام کرے اور جو کوئی اللہ اور آخرت کے دن پر ایمان رکھتا ہو اسے چاہیے کہ خیر کی بات کہے ورنہ خاموش رہے۔
2|عَنْ عُقْبَةَ بْنِ عَامِرٍ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ أَنَّهٗ قَالَ: لَا خَيْرَ فِيْمَنْ لَا يُضِيْفُ.|مسند احمد: رقم الحدیث 17350|لَا=نہیں~خَيْرَ=کوئی خیر~فِيْ=میں~مَنْ=جو~لَا يُضِيْفُ=مہمان نوازی نہیں کرتا|حضرت عقبہ بن عامر رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا کہ جو آدمی مہمان نوازی نہیں کرتا اس میں کوئی بھلائی نہیں ہے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّ مِنَ السُّنَّةِ أَنْ يَخْرُجَ الرَّجُلُ مَعَ ضَيْفِهٖ إِلٰى بَابِ الدَّارِ.|سنن ابن ماجہ: رقم الحدیث 3358|إِنَّ=بلاشبہ~مِنَ=سے~السُّنَّةِ=سنت~أَنْ=کہ~يَخْرُجَ=وہ نکلے~الرَّجُلُ=آدمی~مَعَ=ساتھ~ضَيْفِهٖ=اس کا مہمان~إِلٰى=تک~بَابِ=دروازہ~الدَّارِ=گھر|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: یہ بھی سنت ہے کہ آدمی مہمان کے ساتھ (اسے رخصت کرتے وقت) گھر کے دروازے تک آئے۔
# آٹھویں فصل: استاذ اور شاگرد کے حقوق
1|عَنْ أَبِيْ هَارُوْنَ الْعَبْدِيِّ قَالَ: كُنَّا نَأْتِيْ أَبَا سَعِيْدٍ رَضِيَ اللّٰهُ عَنْهُ فَيَقُوْلُ: مَرْحَبًا بِوَصِيَّةِ رَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ إِنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: إِنَّ النَّاسَ لَكُمْ تَبَعٌ وَإِنَّ رِجَالًا يَأْتُوْنَكُمْ مِنْ أَقْطَارِ الْأَرْضِ يَتَفَقَّهُوْنَ فِي الدِّيْنِ فَإِذَا أَتَوْكُمْ فَاسْتَوْصُوْا بِهِمْ خَيْرًا.|سنن الترمذی: رقم الحدیث 2650|تَبَعٌ=تابع~يَأْتُوْنَكُمْ=وہ تمہارے پاس آئیں گے~أَقْطَارِ=گوشے~يَتَفَقَّهُوْنَ=وہ سمجھنا چاہیں گے~الدِّيْنِ=دین~فَإِذَا=پس جب~أَتَوْكُمْ=وہ تمہارے پاس آئیں گے~فَاسْتَوْصُوْا=پس تم سلوک کرو / تم برتاؤ کرو~خَيْرًا=بھلائی|حضرت ابو ہارون رحمہ اللہ کہتے ہیں کہ ہم ابو سعید خدری رضی اللہ عنہ کے پاس (حصول علم کے لیے) آتے تو وہ کہتے: اللہ کے رسول صلی اللہ علیہ وسلم کی وصیت کے مطابق تمہیں خوش آمدید۔ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: لوگ تمہارے تابع ہیں۔ کچھ لوگ تمہارے پاس زمین کے گوشے گوشے سے علم دین حاصل کرنے کے لیے آئیں گے تو جب وہ تمہارے پاس آئیں تو تمہیں چاہیے کہ ان کے ساتھ بھلائی والا معاملہ کرو۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: إِنَّمَا أَنَا لَكُمْ بِمَنْزِلَةِ الْوَالِدِ أُعَلِّمُكُمْ.|سنن ابی داؤد: رقم الحدیث 8|إِنَّمَا=بلاشبہ~أَنَا=میں~لَكُمْ=تمہارے لیے~بِمَنْزِلَةِ=درجہ میں~الْوَالِدِ=باپ~أُعَلِّمُكُمْ=میں تمہیں سکھاتا ہوں|حضرت ابوہریرہ رضی اللہ عنہ بیان کرتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: بلاشبہ میں تمہارے لیے والد کی مانند ہوں؛ میں تمہیں سکھاتا ہوں۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: تَوَاضَعُوْا لِمَنْ تَعَلَّمُوْنَ مِنْهُ وَتَوَاضَعُوْا لِمَنْ تُعَلِّمُوْنَهٗ وَلَا تَكُوْنُوْا جَبَابِرَةَ الْعُلَمَاءِ.|فیض القدیر شرح الجامع الصغیر: رقم الحدیث 3381|تَوَاضَعُوْا=تم عاجزی اختیار کرو~تَعَلَّمُوْنَ=تم علم حاصل کرتے ہو~تُعَلِّمُوْنَهٗ=اس کو تم سکھاتے ہو~جَبَابِرَةَ=متکبر|حضرت ابوہریرہ رضی اللہ عنہ فرماتے ہیں: جن سے علم حاصل کرتے ہو اور جنہیں سکھاتے ہو اُن کے لیے تواضع اور عاجزی اختیار کرو، متکبر علماء نہ بنو۔
# نویں فصل: عام مسلمانوں کے حقوق
1|عَنْ عَبْدِ اللّٰهِ بْنِ عَمْرٍو رَضِيَ اللّٰهُ عَنْهُمَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: اَلْمُسْلِمُ مَنْ سَلِمَ الْمُسْلِمُوْنَ مِنْ لِسَانِهٖ وَيَدِهٖ وَالْمُهَاجِرُ مَنْ هَجَرَ مَا نَهَى اللّٰهُ عَنْهُ.|صحیح البخاری: رقم الحدیث 10|سَلِمَ=محفوظ رہا~لِسَانِهٖ=اس کی زبان~يَدِهٖ=اس کا ہاتھ~هَجَرَ=اس نے چھوڑ دیا~نَهٰى=منع کیا~اللّٰهُ=اللہ~عَنْهُ=اس سے|حضرت عبد اللہ بن عمرو (بن عاص) رضی اللہ عنہما نبی کریم صلی اللہ علیہ وسلم سے روایت کرتے ہیں کہ آپ صلی اللہ علیہ وسلم نے فرمایا: (حقیقی) مسلمان وہ ہے جس کی زبان اور ہاتھ سے دوسرے مسلمان محفوظ رہیں اور (حقیقی) مہاجر وہ ہے جو ان کاموں کو چھوڑ دے جن سے اللہ تعالیٰ نے منع فرمایا۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: سَمِعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقُوْلُ: حَقُّ الْمُسْلِمِ عَلَى الْمُسْلِمِ خَمْسٌ: رَدُّ السَّلَامِ وَعِيَادَةُ الْمَرِيْضِ وَاتِّبَاعُ الْجَنَائِزِ وَإِجَابَةُ الدَّعْوَةِ وَتَشْمِيْتُ الْعَاطِسِ.|صحیح البخاری: رقم الحدیث 1240|خَمْسٌ=پانچ~رَدُّ السَّلَامِ=سلام کا جواب دینا~عِيَادَةُ=بیمار پرسی کرنا~الْمَرِيْضِ=بیمار~اتِّبَاعُ=پیچھے چلنا~الْجَنَائِزِ=جنازے~إِجَابَةُ=قبول کرنا~الدَّعْوَةِ=دعوت~تَشْمِيْتُ=چھینک کا جواب دینا~الْعَاطِسِ=چھینکنے والا|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ میں نے رسول اللہ صلی اللہ علیہ وسلم سے سنا ہے کہ ایک مسلمان کے دوسرے مسلمان بھائی پر پانچ حقوق ہیں: سلام کا جواب دینا، مریض کی عیادت کرنا، جنازے کے پیچھے چلنا (جنازے میں شرکت کرنا)، دعوت قبول کرنا اور چھینک پر (اس کے اَلْحَمْدُ لِلّٰهِ کہنے کے جواب میں) يَرْحَمُكَ اللّٰهُ کہنا۔
3|عَنْ جَرِيْرِ بْنِ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: بَايَعْتُ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ عَلٰى إِقَامِ الصَّلَاةِ وَإِيْتَاءِ الزَّكَاةِ وَالنُّصْحِ لِكُلِّ مُسْلِمٍ.|صحیح البخاری: رقم الحدیث 57|بَايَعْتُ=میں نے بیعت کی~رَسُوْلَ اللّٰهِ=اللہ کے رسول~إِقَامِ=قائم کرنا~الصَّلَاةِ=نماز~إِيْتَاءِ=دینا~الزَّكَاةِ=زکوٰۃ~النُّصْحِ=خیر خواہی کرنا~لِكُلِّ مُسْلِمٍ=ہر مسلمان کے لیے|حضرت جریر بن عبد اللہ رضی اللہ عنہ سے مروی ہے، فرماتے ہیں: میں نے رسول اللہ صلی اللہ علیہ وسلم سے نماز قائم کرنے، زکوٰۃ ادا کرنے اور ہر مسلمان کی خیر خواہی کرنے پر بیعت کی۔
# دسویں فصل: عام انسانوں کے حقوق
1|عَنْ عَبْدِ اللّٰهِ بْنِ عَمْرٍو رَضِيَ اللّٰهُ عَنْهُمَا عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَنْ قَتَلَ مُعَاهَدًا لَمْ يَرِحْ رَائِحَةَ الْجَنَّةِ وَإِنَّ رِيْحَهَا تُوْجَدُ مِنْ مَسِيْرَةِ أَرْبَعِيْنَ عَامًا.|صحیح البخاری: رقم الحدیث 3166|قَتَلَ=اس نے قتل کیا~مُعَاهَدًا=جس شخص سے وعدہ کر لیا گیا ہو~لَمْ يَرِحْ=نہیں سونگھے گا~رَائِحَةَ=خوشبو~الْجَنَّةِ=جنت~رِيْحَهَا=اس کی خوشبو~تُوْجَدُ=پائی جائے گی~مِنْ=سے~مَسِيْرَةِ=فاصلہ~أَرْبَعِيْنَ=چالیس~عَامًا=سال|حضرت عبد اللہ بن عمرو رضی اللہ عنہما سے مروی ہے کہ نبی کریم صلی اللہ علیہ وسلم نے فرمایا: جس نے کسی معاہد کو (یعنی ایسا غیر مسلم جو مسلمانوں کے ملک میں رہتا ہو اور اس نے معاہدہ کر لیا ہو کہ آپ کے ملک کے قوانین کی پاسداری کروں گا) قتل کیا وہ جنت کی خوشبو بھی نہیں سونگھ سکے گا۔ حالانکہ جنت کی خوشبو چالیس سال کی مسافت سے سونگھی جا سکتی ہے۔
2|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: أَعْطُوا الْأَجِيْرَ أَجْرَهٗ مِنْ قَبْلِ أَنْ يَجِفَّ عَرَقُهٗ.|شرح مشکل الآثار: رقم الحدیث 3014|أَعْطُوا=ادا کر دو~الْأَجِيْرَ=مزدور~أَجْرَهٗ=اس کی مزدوری~يَجِفَّ=سوکھ جائے~عَرَقُهٗ=اس کا پسینہ|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: مزدور کا پسینہ خشک ہونے سے پہلے اس کی مزدوری ادا کر دیا کرو۔
3|عَنْ عَبْدِ اللّٰهِ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَلْخَلْقُ عِيَالُ اللّٰهِ فَأَحَبُّ الْخَلْقِ إِلَى اللّٰهِ مَنْ أَحْسَنَ إِلٰى عِيَالِهٖ.|شعب الایمان: رقم الحدیث 7448|الْخَلْقُ=تمام مخلوق~عِيَالُ=کنبہ~فَأَحَبُّ=سب سے محبوب~أَحْسَنَ=اچھا سلوک کرے~إِلٰى عِيَالِهٖ=اس کے عیال سے|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: ساری مخلوق اللہ تعالیٰ کا کنبہ ہے اس لیے اللہ کو اپنی مخلوق میں سب سے زیادہ محبوب وہ آدمی ہے جو اللہ کی عیال (مخلوق) کے ساتھ اچھا سلوک کرے۔
## اَلْبَابُ الْخَامِسُ: اَلْمُعَامَلَاتُ — معاملات
# پہلی فصل: تجارت
1|عَنْ صَخْرٍ الْغَامِدِيِّ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: اَللّٰهُمَّ بَارِكْ لِأُمَّتِيْ فِيْ بُكُوْرِهَا وَكَانَ إِذَا بَعَثَ سَرِيَّةً أَوْ جَيْشًا بَعَثَهُمْ مِنْ أَوَّلِ النَّهَارِ وَكَانَ صَخْرٌ رَجُلًا تَاجِرًا وَكَانَ يَبْعَثُ تِجَارَتَهٗ مِنْ أَوَّلِ النَّهَارِ فَأَثْرٰى وَكَثُرَ مَالُهٗ.|سنن ابی داؤد: رقم الحدیث 2606|اَللّٰهُمَّ=اے اللہ~بَارِكْ=برکت عطا فرما~لِأُمَّتِيْ=میری امت کے لیے~فِيْ=میں~بُكُوْرِهَا=اس کے ابتدائی وقت~كَانَ=تھا~إِذَا=جب~بَعَثَ=اس نے بھیجا~سَرِيَّةً=کوئی سریہ~أَوْ=یا~جَيْشًا=لشکر~بَعَثَهُمْ=اس نے ان کو بھیجا~مِنْ أَوَّلِ=پہلا~النَّهَارِ=دن کا وقت~صَخْرٌ=صخر (ایک صحابی کا نام)~رَجُلًا=آدمی~تَاجِرًا=تاجر~يَبْعَثُ=وہ بھیجتا ہے~تِجَارَتَهٗ=اس کی تجارت~فَأَثْرٰى=پس وہ مالدار ہو گیا~كَثُرَ=بہت ہو گیا~مَالُهٗ=اس کا مال|حضرت صخر غامدی رضی اللہ عنہ کہتے ہیں کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: اَللّٰهُمَّ بَارِكْ لِأُمَّتِيْ فِيْ بُكُوْرِهَا (اے اللہ! میری امت کے لیے دن کے ابتدائی حصہ میں برکت دے) اور جب بھی آپ صلی اللہ علیہ وسلم کوئی سریہ یا لشکر روانہ کرتے تو دن کی ابتدا میں بھیجتے۔ (راوی کہتے ہیں) حضرت صخر غامدی رضی اللہ عنہ تجارت کیا کرتے تھے۔ وہ اپنی تجارت کی ابتدا صبح سویرے کرتے تھے تو (اس کی برکت سے) وہ مالدار ہو گئے اور ان کا مال بہت بڑھ گیا۔
2|عَنْ إِسْمَاعِيْلَ بْنِ عُبَيْدِ بْنِ رِفَاعَةَ عَنْ أَبِيْهِ عَنْ جَدِّهٖ رَضِيَ اللّٰهُ عَنْهُ أَنَّهٗ خَرَجَ مَعَ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ إِلَى الْمُصَلّٰى فَرَأَى النَّاسَ يَتَبَايَعُوْنَ فَقَالَ: يَا مَعْشَرَ التُّجَّارِ! فَاسْتَجَابُوْا لِرَسُوْلِ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ وَرَفَعُوْا أَعْنَاقَهُمْ وَأَبْصَارَهُمْ إِلَيْهِ فَقَالَ: إِنَّ التُّجَّارَ يُبْعَثُوْنَ يَوْمَ الْقِيَامَةِ فُجَّارًا إِلَّا مَنِ اتَّقَى اللّٰهَ وَبَرَّ وَصَدَقَ.|سنن الترمذی: رقم الحدیث 1210|خَرَجَ=وہ نکلا~مَعَ=ساتھ~النَّبِيِّ=نبی~إِلَى الْمُصَلّٰى=عید گاہ کی طرف~فَرَأَى=پس اس نے دیکھا~النَّاسَ=لوگ~يَتَبَايَعُوْنَ=وہ خرید و فروخت کرتے ہیں~فَقَالَ=پس فرمایا~يَا=اے~مَعْشَرَ=جماعت~التُّجَّارِ=تجارت کرنے والے~فَاسْتَجَابُوْا=پس وہ متوجہ ہوئے~لِرَسُوْلِ اللّٰهِ=اللہ کے رسول کے لیے~رَفَعُوْا=انہوں نے بلند کیا~أَعْنَاقَهُمْ=اپنی گردنیں~أَبْصَارَهُمْ=اپنی نظریں~إِلَيْهِ=اس کی طرف~يُبْعَثُوْنَ=وہ اٹھائے جائیں گے~يَوْمَ الْقِيَامَةِ=قیامت کے دن~فُجَّارًا=گنہگار~إِلَّا=مگر~مَنِ=جو~اتَّقَى=وہ ڈرا~اللّٰهَ=اللہ سے~بَرَّ=اس نے نیکی کی~صَدَقَ=اس نے سچ بولا|حضرت رفاعہ رضی اللہ عنہ سے مروی ہے کہ وہ نبی اکرم صلی اللہ علیہ وسلم کے ساتھ عید گاہ کی طرف نکلے، آپ صلی اللہ علیہ وسلم نے لوگوں کو خرید و فروخت کرتے ہوئے دیکھا تو فرمایا: اے تاجروں کے گروہ! تو لوگوں نے نبی اکرم صلی اللہ علیہ وسلم کی بات کو سننے کے لیے اپنی گردنیں آپ کی طرف بلند کیں اور نظروں کو اونچا کر لیا (یعنی خوب متوجہ ہو گئے) تو آپ صلی اللہ علیہ وسلم نے فرمایا: بے شک تاجروں کو قیامت کے دن گنہگار اٹھایا جائے گا سوائے اس تاجر کے جو اللہ سے ڈرے، نیک کام کرے اور سچ بولے۔
3|عَنِ ابْنِ عُمَرَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَلتَّاجِرُ الْأَمِيْنُ الصَّدُوْقُ الْمُسْلِمُ مَعَ الشُّهَدَاءِ يَوْمَ الْقِيَامَةِ.|سنن ابن ماجہ: رقم الحدیث 2139|التَّاجِرُ=تجارت کرنے والا~الْأَمِيْنُ=امانت دار~الصَّدُوْقُ=سچا~الْمُسْلِمُ=مسلمان~مَعَ=ساتھ~الشُّهَدَاءِ=شہید کی جمع~يَوْمَ الْقِيَامَةِ=قیامت کے دن|حضرت عبد اللہ بن عمر رضی اللہ عنہما کہتے ہیں کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: امانت دار اور سچا مسلمان تاجر قیامت کے دن شہداء کے ساتھ ہوگا۔
# دوسری فصل: سود
1|عَنْ أَبِيْ سَعِيْدٍ الْخُدْرِيِّ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: اَلذَّهَبُ بِالذَّهَبِ وَالْفِضَّةُ بِالْفِضَّةِ وَالْبُرُّ بِالْبُرِّ وَالشَّعِيْرُ بِالشَّعِيْرِ وَالتَّمْرُ بِالتَّمْرِ وَالْمِلْحُ بِالْمِلْحِ مِثْلًا بِمِثْلٍ يَدًا بِيَدٍ فَمَنْ زَادَ أَوِ اسْتَزَادَ فَقَدْ أَرْبٰى الْآخِذُ وَالْمُعْطِيْ فِيْهِ سَوَاءٌ.|صحیح مسلم: رقم الحدیث 4064|الذَّهَبُ=سونا~بِالذَّهَبِ=سونے کے بدلے~الْفِضَّةُ=چاندی~الْبُرُّ=گندم~الشَّعِيْرُ=جو~التَّمْرُ=کھجور~الْمِلْحُ=نمک~مِثْلًا بِمِثْلٍ=برابر سرابر~يَدًا بِيَدٍ=نقد بنقد~زَادَ=اس نے زیادہ دیا~اسْتَزَادَ=اس نے زیادتی کا مطالبہ کیا~أَرْبٰى=اس نے سود لیا~الْآخِذُ=لینے والا~الْمُعْطِيْ=دینے والا|حضرت ابو سعید خدری رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا کہ سونا سونے کے بدلے، چاندی چاندی کے بدلے، گندم گندم کے بدلے، جو جو کے بدلے، کھجور کھجور کے بدلے، نمک نمک کے بدلے، برابر سرابر اور نقد در نقد فروخت کیا کرو۔ جس آدمی نے زیادہ دیا یا زیادتی کا مطالبہ کیا تو اس نے سودی کاروبار کیا۔ اس میں زیادہ لینے اور زیادہ دینے والا؛ دونوں گناہ میں برابر شریک ہیں۔
2|عَنِ ابْنِ مَسْعُوْدٍ رَضِيَ اللّٰهُ عَنْهُ قَالَ: لَعَنَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ آكِلَ الرِّبَا وَمُؤْكِلَهٗ وَشَاهِدَيْهِ وَكَاتِبَهٗ.|سنن الترمذی: رقم الحدیث 1206|لَعَنَ=لعنت کی~آكِلَ=کھانے والا~الرِّبَا=سود~مُؤْكِلَهٗ=اس کا کھلانے والا~شَاهِدَيْهِ=اس کے دونوں گواہ~كَاتِبَهٗ=اس کا لکھنے والا|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے سود لینے والے، سود دینے والے، سود کے دونوں گواہوں اور اس (سودی معاملہ) کے لکھنے والے پر لعنت فرمائی ہے۔
3|عَنِ ابْنِ مَسْعُوْدٍ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَا أَحَدٌ أَكْثَرَ مِنَ الرِّبَا إِلَّا كَانَ عَاقِبَةُ أَمْرِهٖ إِلٰى قِلَّةٍ.|سنن ابن ماجہ: رقم الحدیث 2279|مَا=نہیں~أَحَدٌ=کوئی~أَكْثَرَ=اس نے زیادہ کیا~مِنَ الرِّبَا=سود سے~إِلَّا=مگر~عَاقِبَةُ=انجام~أَمْرِهٖ=اس کا کام~إِلٰى=طرف~قِلَّةٍ=کمی|حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: جو آدمی بھی سود سے مال بڑھائے گا تو اس کا نتیجہ یہ ہوتا ہے کہ اس کا مال کم ہی ہو جاتا ہے۔
# تیسری فصل: مضاربت و شرکت
1|حَدَّثَنَا ابْنُ عَوْنٍ رَحِمَهُ اللّٰهُ قَالَ: كَانَ مُحَمَّدٌ رَحِمَهُ اللّٰهُ يَقُوْلُ: اَلْأَرْضُ عِنْدِيْ مِثْلُ مَالِ الْمُضَارَبَةِ فَمَا صَلَحَ فِيْ مَالِ الْمُضَارَبَةِ صَلَحَ فِي الْأَرْضِ وَمَا لَمْ يَصْلُحْ فِيْ مَالِ الْمُضَارَبَةِ لَمْ يَصْلُحْ فِي الْأَرْضِ قَالَ: وَكَانَ لَا يَرٰى بَأْسًا أَنْ يَدْفَعَ أَرْضَهٗ كُلَّهَا إِلَى الْأَكَّارِ عَلٰى أَنْ يَعْمَلَ فِيْهَا بِنَفْسِهٖ وَوَلَدِهٖ وَأَعْوَانِهٖ وَبَقَرِهٖ وَلَا يُنْفِقُ شَيْئًا وَتَكُوْنُ النَّفَقَةُ كُلُّهَا مِنْ رَبِّ الْأَرْضِ.|سنن النسائی: رقم الحدیث 3960|الْأَرْضُ=زمین~عِنْدِيْ=میرے نزدیک~مِثْلُ=طرح~مَالِ=مال~الْمُضَارَبَةِ=مضاربت~فَمَا=پس جو کچھ~صَلَحَ=درست ہے~فِيْ=میں~لَمْ يَصْلُحْ=وہ درست نہیں ہے~لَا يَرٰى=نہیں دیکھتا تھا~بَأْسًا=کوئی حرج~أَنْ يَدْفَعَ=کہ دے دے~أَرْضَهٗ=اس کی زمین~الْأَكَّارِ=مزارع~يَعْمَلَ=وہ عمل کرے~فِيْهَا=اس میں~بِنَفْسِهٖ=بذات خود~وَلَدِهٖ=اس کا بیٹا~أَعْوَانِهٖ=اس کے مددگار~بَقَرِهٖ=اس کی گائے~لَا يُنْفِقُ=وہ خرچ نہ کرے~شَيْئًا=کوئی چیز~النَّفَقَةُ=خرچ~كُلُّهَا=اس کا تمام~رَبِّ الْأَرْضِ=مالک زمین|حضرت ابن عون رحمہ اللہ کہتے ہیں کہ حضرت محمد بن سیرین رحمہ اللہ فرماتے تھے کہ میرے نزدیک زمین (بٹائی پر دینا) مضاربت کے مال کی طرح ہے۔ جو کچھ مال مضاربت میں جائز ہے وہ زمین (بٹائی پر دینے) میں بھی جائز ہے اور جو مال مضاربت میں جائز نہیں وہ زمین میں بھی جائز نہیں۔ اور وہ اس بات میں کوئی حرج نہیں سمجھتے تھے کہ زمین مزارع کے سپرد کر دے اور وہ (مزارع) اس میں خود یا اپنی اولاد یا اپنے ساتھیوں اور اپنے بیلوں وغیرہ کے ساتھ کام کرے اور خرچ کچھ نہ کرے بلکہ اخراجات سب کے سب مالک زمین کی طرف سے ہوں۔
2|عَنِ السَّائِبِ رَضِيَ اللّٰهُ عَنْهُ قَالَ لِلنَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: كُنْتَ شَرِيْكِيْ فِي الْجَاهِلِيَّةِ فَكُنْتَ خَيْرَ شَرِيْكٍ كُنْتَ لَا تُدَارِيْنِيْ وَلَا تُمَارِيْنِيْ.|سنن ابن ماجہ: رقم الحدیث 2287|كُنْتَ=آپ تھے~شَرِيْكِيْ=میرے ساتھی~فِي الْجَاهِلِيَّةِ=زمانہ جاہلیت میں~خَيْرَ=بہترین~شَرِيْكٍ=ساتھی~لَا تُدَارِيْنِيْ=آپ میری مخالفت نہیں کرتے تھے~لَا تُمَارِيْنِيْ=آپ مجھ سے جھگڑتے نہیں تھے|حضرت سائب رضی اللہ عنہ کہتے ہیں کہ انہوں نے نبی اکرم صلی اللہ علیہ وسلم سے عرض کیا: (اے اللہ کے رسول!) زمانہ جاہلیت میں آپ میرے ساتھ تجارت میں شریک تھے تو آپ شراکت میں بہت بہترین ثابت ہوئے۔ نہ تو آپ میری مخالفت کرتے تھے اور نہ ہی آپ جھگڑتے تھے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ رَفَعَهٗ قَالَ: إِنَّ اللّٰهَ يَقُوْلُ: أَنَا ثَالِثُ الشَّرِيْكَيْنِ مَا لَمْ يَخُنْ أَحَدُهُمَا صَاحِبَهٗ فَإِذَا خَانَهٗ خَرَجْتُ مِنْ بَيْنِهِمْ.|سنن ابی داؤد: رقم الحدیث 3383|أَنَا=میں~ثَالِثُ=تیسرا~الشَّرِيْكَيْنِ=دو شریکوں~مَا لَمْ يَخُنْ=یعنی جب تک کہ خیانت نہ کرے~أَحَدُهُمَا=ان دو میں سے ایک~صَاحِبَهٗ=اس کا ساتھی~فَإِذَا=پس جب~خَانَهٗ=اس نے اس سے خیانت کی~خَرَجْتُ=میں نکل جاتا ہوں~مِنْ=سے~بَيْنِهِمْ=ان کے درمیان|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے وہ مرفوعاً روایت نقل کرتے ہیں کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: اللہ تعالیٰ فرماتے ہیں: میں (کاروبار کے) دو شریکوں کا تیسرا ہوں جب تک کہ ان دونوں شریکوں میں سے کوئی ایک شریک دوسرے ساتھی کے ساتھ خیانت نہ کرے اور جب کوئی شریک اپنے ساتھی کے ساتھ خیانت کرتا ہے تو میں ان کے درمیان سے نکل جاتا ہوں (میری مدد ختم ہو جاتی ہے)۔
# چوتھی فصل: ہدیہ
1|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ قَالَ: قَالَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ: تَهَادُوْا تَحَابُّوْا.|شعب الایمان: رقم الحدیث 8976|تَهَادُوْا=ایک دوسرے کو ہدیہ دو~تَحَابُّوْا=تم آپس میں محبت کرنے لگو گے|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے فرمایا: ایک دوسرے کو ہدیہ دیا کرو اس سے آپس میں محبت بڑھتی ہے۔
2|عَنْ عَائِشَةَ رَضِيَ اللّٰهُ عَنْهَا قَالَتْ: كَانَ رَسُوْلُ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ يَقْبَلُ الْهَدِيَّةَ وَيُثِيْبُ عَلَيْهَا.|صحیح البخاری: رقم الحدیث 2585|يَقْبَلُ=وہ قبول کیا کرتے تھے~الْهَدِيَّةَ=ہدیہ / تحفہ~يُثِيْبُ=بدلہ دیا کرتے تھے~عَلَيْهَا=اس پر|حضرت عائشہ رضی اللہ عنہا سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم ہدیہ قبول بھی فرمایا کرتے تھے اور اس کا بدلہ بھی دیا کرتے تھے۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَوْ دُعِيْتُ إِلٰى ذِرَاعٍ أَوْ كُرَاعٍ لَأَجَبْتُ وَلَوْ أُهْدِيَ إِلَيَّ ذِرَاعٌ أَوْ كُرَاعٌ لَقَبِلْتُ.|صحیح البخاری: رقم الحدیث 2568|لَوْ=اگر~دُعِيْتُ=مجھے دعوت دی جائے~إِلٰى=طرف~ذِرَاعٍ=دستی~أَوْ=یا~كُرَاعٍ=پائے~لَأَجَبْتُ=تو میں قبول کر لوں گا~أُهْدِيَ=ہدیہ دیا جائے~إِلَيَّ=میری طرف~لَقَبِلْتُ=تو میں قبول کر لوں گا|حضرت ابوہریرہ رضی اللہ عنہ سے مروی ہے کہ نبی کریم صلی اللہ علیہ وسلم نے فرمایا: اگر مجھے (جانور کی) دستی اور پائے (کے گوشت) کی بھی دعوت دی جائے تو میں قبول کر لوں گا اور اگر میری طرف (جانور کی) دستی یا پائے (کے گوشت) کا ہدیہ بھی بھیجا جائے تو میں اسے بھی قبول کر لوں گا۔
# پانچویں فصل: وصیت و وراثت
1|عَنِ ابْنِ عُمَرَ رَضِيَ اللّٰهُ عَنْهُمَا أَنَّ رَسُوْلَ اللّٰهِ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: مَا حَقُّ امْرِئٍ مُسْلِمٍ لَهٗ شَيْءٌ يُوْصِيْ فِيْهِ يَبِيْتُ لَيْلَتَيْنِ إِلَّا وَوَصِيَّتُهٗ مَكْتُوْبَةٌ عِنْدَهٗ.|سنن الترمذی: رقم الحدیث 974|مَا حَقُّ=نہیں حق~امْرِئٍ=آدمی~مُسْلِمٍ=مسلمان~لَهٗ=اس کے لیے~شَيْءٌ=کوئی چیز~يُوْصِيْ=وصیت کرے~يَبِيْتُ=وہ گزارے~لَيْلَتَيْنِ=دو راتیں~وَصِيَّتُهٗ=اس کی وصیت~مَكْتُوْبَةٌ=لکھی ہوئی~عِنْدَهٗ=اس کے پاس|حضرت عبد اللہ بن عمر رضی اللہ عنہما سے روایت ہے کہ رسول اللہ صلی اللہ علیہ وسلم نے ارشاد فرمایا: جس مسلمان بندے کی دو راتیں بھی اس حال میں گزریں کہ اس کے پاس کوئی ایسی چیز ہو جس کے بارے میں وصیت کرنی چاہیے تو اس کے ذمے ضروری ہے کہ اس کے پاس اس کا وصیت نامہ لکھا ہوا موجود ہو۔
2|عَنْ أُسَامَةَ بْنِ زَيْدٍ رَضِيَ اللّٰهُ عَنْهُمَا أَنَّ النَّبِيَّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: لَا يَرِثُ الْمُسْلِمُ الْكَافِرَ وَلَا الْكَافِرُ الْمُسْلِمَ.|صحیح البخاری: رقم الحدیث 6764|لَا يَرِثُ=وارث نہیں بنے گا~الْمُسْلِمُ=مسلمان~الْكَافِرَ=کافر کا~لَا=اور نہ ہی~الْكَافِرُ=کافر~الْمُسْلِمَ=مسلمان کا|حضرت اسامہ بن زید رضی اللہ عنہما سے مروی ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے ارشاد فرمایا: مسلمان؛ کافر کا وارث نہیں بنے گا اور نہ ہی کافر؛ مسلمان کا وارث بنے گا۔
3|عَنْ أَبِيْ هُرَيْرَةَ رَضِيَ اللّٰهُ عَنْهُ عَنِ النَّبِيِّ صَلَّى اللّٰهُ عَلَيْهِ وَسَلَّمَ قَالَ: اَلْقَاتِلُ لَا يَرِثُ.|سنن الترمذی: رقم الحدیث 2109|الْقَاتِلُ=قتل کرنے والا~لَا يَرِثُ=وارث نہیں ہوگا|حضرت ابوہریرہ رضی اللہ عنہ سے روایت ہے کہ نبی اکرم صلی اللہ علیہ وسلم نے فرمایا: قاتل (مقتول کا) وارث نہیں ہوگا۔
